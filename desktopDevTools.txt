  // Data-stability tracking. Tab-click detection only proves a tab's fetch was triggered, not that
  // its data has actually finished arriving and merging — TwinSpires keeps sending per-horse field
  // updates in the background after that. Fingerprinting the actual scoring fields and requiring
  // them to hold steady for a few seconds catches that, instead of trusting tab clicks alone.
  let lastFieldFingerprint = null;
  let fieldStableSince = null;
  const FIELD_STABILITY_WINDOW_MS = 6000;

  function computeFieldFingerprint() {
    let liveHorses = Object.values(cachedHorsesMap)
      .filter(h => !h.IS_SCRATCHED && !cachedScratchedMap[h.PROGRAM])
      .sort((a, b) => String(a.PROGRAM).localeCompare(String(b.PROGRAM), undefined, { numeric: true }));
    return liveHorses.map(h => SCORING_NUM_FIELDS.map(f => h[f]).join(',')).join('|');
  }

  // Updates the fingerprint/timer as a side effect, then reports whether the data has been
  // unchanged for at least FIELD_STABILITY_WINDOW_MS. Call once per render pass.
  function checkFieldStability() {
    let fp = computeFieldFingerprint();
    if (fp !== lastFieldFingerprint) {
      lastFieldFingerprint = fp;
      fieldStableSince = Date.now();
      return false;
    }
    return fieldStableSince !== null && (Date.now() - fieldStableSince) >= FIELD_STABILITY_WINDOW_MS;
  }

  // TwinSpires has no Summary tab for harness races, so it is neither waited on nor listed when the breed is Harness.
  function getRequiredTabs() {
    let isHarness = String(cachedBreed || '').toLowerCase() === 'harness';
    return isHarness ? REQUIRED_TABS.filter(tab => tab !== 'Summary') : REQUIRED_TABS;
  }

  function getLoadingProgress() {
    let hasComments = Object.values(cachedHorsesMap).some(h => 
      (h.COMMENTS_POS && h.COMMENTS_POS !== '""') || 
      (h.COMMENTS_NEG && h.COMMENTS_NEG !== '""')
    );
    if (hasComments) {
      loadedTabs['Comments'] = true;
    }

    if (!loadedTabs['Comments'] && (Date.now() - collectionStartTime) >= 15000) {
      loadedTabs['Comments'] = true;
    }

    let tabs = getRequiredTabs();
    let completed = 0;
    tabs.forEach(tab => {
      if (loadedTabs[tab]) completed++;
    });
    let percent = Math.round((completed / tabs.length) * 100);
    return { completed, total: tabs.length, percent };
  }

  function applyWidth(keepRightEdge) {
    let overlay = document.getElementById('ts-model-overlay');
    if (overlay) {
      let oldW = overlay.offsetWidth;
      overlay.style.width = `min(${currentWidth}px, 94vw)`;
      // If the panel was moved (positioned by `left`), keep its visual right edge fixed on resize.
      if (keepRightEdge && overlay.style.right === 'auto') {
        let newW = overlay.offsetWidth;
        let curLeft = parseFloat(overlay.style.left) || 0;
        overlay.style.left = `${curLeft + (oldW - newW)}px`;
      }
      let label = document.getElementById('ts-width-label');
      if (label) label.innerText = `${currentWidth}px`;
      let slider = document.getElementById('ts-width-slider');
      if (slider && parseFloat(slider.value) !== currentWidth) slider.value = currentWidth;
    }
    removeZoomPreview();
  }

  // Race/track change detection. Compares a normalized key (track|raceNum) that ONLY this function
  // and the Reset button write — payload parsing never touches it. (The old check compared the URL
  // against cachedRaceNum/cachedTrackName, but every incoming payload overwrites those, so by the
  // time a check ran the "previous" value could already equal the new one and the reset was
  // silently skipped, leaving stale Comments/Pools behind.) Comparison is normalized so 5 vs "5"
  // or differing capitalization/punctuation in the track name don't count as a change.
  let activeRaceKey = null;
  function makeRaceKey(track, race, prevKey) {
    let prev = prevKey ? prevKey.split('|') : ['', ''];
    let t = (track && track !== 'Unknown Track') ? String(track).toLowerCase().replace(/[^a-z0-9]/g, '') : '';
    let r = race ? String(race).replace(/[^0-9]/g, '') : '';
    return (t || prev[0]) + '|' + (r || prev[1]);
  }
  function checkRaceChange(track, race) {
    let newKey = makeRaceKey(track, race, activeRaceKey);
    if (newKey === '|') return false;
    if (activeRaceKey === null) { // first sighting this session — record it, nothing to reset yet
      activeRaceKey = newKey;
      return false;
    }
    if (newKey === activeRaceKey) return false;
    resetRaceData();
    activeRaceKey = newKey;
    if (track && track !== 'Unknown Track') cachedTrackName = track;
    if (race) cachedRaceNum = race;
    updateOverlay();
    return true;
  }

  function parseTwinSpiresUrl(url = window.location.href) {

    const match = url.match(regex);
    if (match) {
      let trackSlug = match[1];
      let trackCode = match[2];
      let breed = match[3];
      let raceNum = parseInt(match[4], 10);
      let trackName = trackSlug.split('-').map(word => word.charAt(0).toUpperCase() + word.slice(1)).join(' ');
      return { trackName, trackCode, breed, raceNum };
    }
    return null;
  }

  // Shared takeout assumption used to convert pool-distribution $ amounts into real payouts,
  // and as the fallback exacta payout estimator's discount. Validated against a real UK Haydock
  // WIN pool (~18% reproduced the track's displayed odds almost exactly).
  const TAKEOUT_RATE = 0.18;

  // Race-specific takeout backed out of the WIN pool. Displayed odds = pool x (1 - takeout) / horse$ - 1,
  // so takeout = 1 - (odds + 1) x horse$ / pool, computed per active horse (live odds only — morning
  // line isn't pool-based) and the MEDIAN taken (smooths breakage rounding and max-odds-capped
  // horses). Needs >= 3 usable horses and a result within 10-30%, otherwise falls back to the
  // TAKEOUT_RATE constant. Place/Show have no displayed odds to solve from, so they borrow this
  // figure (assumes straight pools share a rate — not verified). Trifecta cap stays on the constant.
  function getDerivedTakeout() {
    if (!cachedWinPoolTotal || cachedWinPoolTotal <= 0) return { rate: null, n: 0 };
    let vals = [];
    Object.values(cachedHorsesMap).forEach(h => {
      if (h.IS_SCRATCHED || cachedScratchedMap[h.PROGRAM]) return;
      let entry = cachedPoolsMap[String(h.PROGRAM)];
      if (!entry || !(entry.win > 0)) return;
      let liveDec = parseOddsToDecimal(h.LIVE_ODDS);
      if (isNaN(liveDec) || liveDec <= 0) return;
      vals.push(1 - (liveDec + 1) * entry.win / cachedWinPoolTotal);
    });
    if (vals.length < 3) return { rate: null, n: 0 };
    vals.sort((a, b) => a - b);
    let mid = Math.floor(vals.length / 2);
    let median = vals.length % 2 ? vals[mid] : (vals[mid - 1] + vals[mid]) / 2;
    if (!(median >= 0.10 && median <= 0.30)) return { rate: null, n: 0 };
    return { rate: median, n: vals.length };
  }
  function getEffectiveTakeout() {
    let d = getDerivedTakeout();
    return d.rate !== null ? d.rate : TAKEOUT_RATE;
  }

  // A bet must promise at least this much back per dollar risked (net of the stake itself) to be

  const TRIFECTA_MIN_BASE_UNIT = 0.5;

  // Returns wetness tier: 0 = dry/fast/firm, 1 = off/yielding (moderately wet), 2 = sloppy/muddy/heavy (fully wet)

  function parseExportDateTime(rawPostTime) {
    let dateObj = new Date();
    let timeStr = "";

    if (rawPostTime && rawPostTime !== "N/A") {
      let parsed = new Date(rawPostTime);
      if (!isNaN(parsed.getTime())) {
        dateObj = parsed;
        let hours = dateObj.getHours();
        let minutes = dateObj.getMinutes();
        let ampm = hours >= 12 ? 'pm' : 'am';
        hours = hours % 12;
        hours = hours ? hours : 12;
        minutes = minutes < 10 ? '0' + minutes : minutes;
        timeStr = `${hours}:${minutes}${ampm}`;
      } else {
        timeStr = rawPostTime.toLowerCase().replace(/\s+/g, '');
      }
    }

    if (!timeStr) {
      let hours = dateObj.getHours();
      let minutes = dateObj.getMinutes();
      let ampm = hours >= 12 ? 'pm' : 'am';
      hours = hours % 12;
      hours = hours ? hours : 12;
      minutes = minutes < 10 ? '0' + minutes : minutes;
      timeStr = `${hours}:${minutes}${ampm}`;
    }

    let dateStr = `${dateObj.getMonth() + 1}/${dateObj.getDate()}/${dateObj.getFullYear()}`;

  }

  function getVal(obj, keys, defaultVal = 0) {
    if (!obj) return defaultVal;
    let keyList = Array.isArray(keys) ? keys : [keys];
    for (let k of keyList) {
      let val = obj[k];
      if (val !== undefined && val !== null && typeof val !== 'object') {
        if (typeof val === 'number') return val;
        let cleaned = String(val).replace('%', '').trim();
        let parsed = parseFloat(cleaned);
        if (!isNaN(parsed)) return parsed;
      }
    }
    return defaultVal;
  }

  function getStr(obj, keys, defaultVal = '') {
    if (!obj) return defaultVal;
    let keyList = Array.isArray(keys) ? keys : [keys];
    for (let k of keyList) {
      let val = obj[k];
      if (val === undefined || val === null) continue;

      // Some fields (e.g. commentsPositive/commentsNegative) legitimately arrive as an ARRAY of
      // strings rather than one flat string — join them into a readable single string. An empty
      // array (a horse with no negative comments, say) correctly falls through to the next
      // candidate key / the default, rather than being treated as present-but-blank.
      if (Array.isArray(val)) {
        let joined = val
          .filter(v => v !== null && v !== undefined && String(v).trim() !== '')
          .map(v => String(v).trim())
          .join('; ');
        if (joined !== '') return joined;
        continue;
      }

      // Guard against object-typed payload fields (e.g. a structured {numerator, denominator}
      // odds object on some responses instead of a flat "7/5" string) — String() on a plain
      // object produces the literal text "[object Object]", which would otherwise silently
      // overwrite a perfectly good previously-merged value. Skip and try the next candidate key.
      if (typeof val !== 'object') {
        return String(val).trim();
      }
    }
    return defaultVal;
  }

  function parseOddsToDecimal(oddsVal) {
    if (!oddsVal) return NaN;
    let s = String(oddsVal).trim();
    if (!s || s.toUpperCase() === 'NAN') return NaN;
    try {
      if (s.includes('/')) {
        let parts = s.split('/');
        return parseFloat(parts[0]) / parseFloat(parts[1]);
      }
      return parseFloat(s);
    } catch (e) {
      return NaN;
    }
  }

          let found = findStatsObject(obj[key], depth + 1);
          if (found) return found;
        }
      }
    }
    return null;
  }

  // Parses the Exotics payload into a { grid, poolTotal, baseUnit, poolType } object.
  // grid[fromProgram][toProgram] = real $ payout per baseUnit stake, or null if self/scratched (-1.00 / -2.00 sentinels).
  function parseExoticsGrid(payload) {
    let grid = {};
    let modelsArr = payload.poolTotalsModels || [];
    // poolTotalsModels can contain multiple pool types together (EX, TR, etc. all as siblings in
    // the same array) — find each by its own PoolType rather than assuming index [0] is "the" entry.
    let exRow = modelsArr.find(m => getStr(m, 'PoolType', '') === 'EX') || modelsArr[0] || {};
    let trRow = modelsArr.find(m => getStr(m, 'PoolType', '') === 'TR') || {};

    let poolTotal = getVal(exRow, 'Amount', 0);
    let baseUnit = getVal(exRow, 'Base', 1) || 1;
    let poolType = getStr(payload, 'poolType', 'EX');

    let trifectaPoolTotal = getVal(trRow, 'Amount', 0);
    let trifectaBaseUnit = getVal(trRow, 'Base', 1) || 1;

    let runList = (payload.Exotics && payload.Exotics.RunList) || [];
    runList.forEach(row => {
      let from = String(row.ProgramNumber);
      grid[from] = grid[from] || {};
      (row.Combos || []).forEach(c => {
        let to = String(c.With);
        let amt = parseFloat(c.Amount);
        grid[from][to] = (isNaN(amt) || amt < 0) ? null : amt;
      });
    });

    return { grid, poolTotal, baseUnit, poolType, trifectaPoolTotal, trifectaBaseUnit };
  }

  // A displayed pari-mutuel payout is pool x (1 - takeout) / money on that combo, and it does not include a bet you
  // have not placed yet. Placing it adds your stake to the combo (addedToCombo) and to the whole pool (addedToPool),
  // which lowers the payout. The money already on the combo is backed out of the displayed payout, then your stake is
  // added to both. Returns the payout unchanged when there is nothing to add or the pool total is unknown. The effect
  // is negligible for a small bet in a big pool and grows as the bet becomes a larger share of the pool.
  function adjustPayoutForOwnStake(payoutPerDollar, poolTotal, takeout, addedToCombo, addedToPool) {
    if (!(payoutPerDollar > 0) || !(poolTotal > 0) || !(addedToCombo > 0) || !(addedToPool > 0)) return payoutPerDollar;
    let comboMoney = poolTotal * (1 - takeout) / payoutPerDollar;
    return (poolTotal + addedToPool) * (1 - takeout) / (comboMoney + addedToCombo);
  }

  // Real per-$1 exacta payout for "from" finishing 1st and "to" finishing 2nd, or null if unavailable.
  // unitStake / ticketTotal (optional): the bettor's own money on this combo / on the whole ticket. When given, the
  // payout is re-priced as if that money were already in the pool (see adjustPayoutForOwnStake).
  function getExactaPayout(from, to, unitStake, ticketTotal) {
    let f = String(from), t = String(to);
    if (cachedExactaGrid[f] && cachedExactaGrid[f][t] !== undefined && cachedExactaGrid[f][t] !== null) {
      // grid values are per cachedExactaBaseUnit stake; normalize to a true per-$1 rate
      let perDollar = cachedExactaGrid[f][t] / (cachedExactaBaseUnit || 1);
      return adjustPayoutForOwnStake(perDollar, cachedExactaPoolTotal, getEffectiveTakeout(), unitStake, ticketTotal);
    }
    return null;
  }

  // Finds the WPSPools payload (Win/Place/Show pool distribution from the Pools tab)
  function findPoolsPayload(obj, depth = 0) {
    if (!obj || typeof obj !== 'object' || depth > 5) return null;
    if (obj.WPSPools && Array.isArray(obj.WPSPools.Entries)) return obj;
    if (Array.isArray(obj)) {
      for (let item of obj) {
        let found = findPoolsPayload(item, depth + 1);
        if (found) return found;
      }
    } else {
      for (let key in obj) {
        if (typeof obj[key] === 'object' && obj[key] !== null) {

          if (found) return found;
        }
      }
    }
    return null;
  }

  // Parses the WPSPools payload. Win/Place/Show are $ amounts WAGERED on each horse (pool share),
  // NOT payouts — confirmed by summing to the pool totals. Real payouts are derived separately
  // via the pari-mutuel formula in getWinPoolPayout/getPlacePoolPayout below.
  // Detects whether a pool's per-horse values are dollar amounts or percentages of the pool,
  // purely from the data itself — dollar amounts across the field should sum close to the real
  // pool total (usually thousands), percentages should sum close to 100. Whichever hypothesis
  // fits better wins. Returns the values converted to dollar-equivalents either way, so downstream
  // math (which assumes dollars) works correctly regardless of which display mode the page was in
  // when the data was captured.
  function normalizePoolValues(rawValues, poolTotal) {
    let validSum = rawValues.filter(v => !isNaN(v) && v > 0).reduce((a, b) => a + b, 0);
    if (validSum <= 0 || !poolTotal) return { values: rawValues, wasPercent: false };
    let distFromPercent = Math.abs(validSum - 100);
    let distFromDollar = Math.abs(validSum - poolTotal);
    let looksLikePercent = distFromPercent < distFromDollar;
    if (!looksLikePercent) return { values: rawValues, wasPercent: false };
    return { values: rawValues.map(v => (isNaN(v) || v <= 0) ? v : (v / 100) * poolTotal), wasPercent: true };
  }

  function parsePoolsPayload(payload) {
    let poolTotalsArr = payload.PoolTotals || [];
    let winTotal = 0, placeTotal = 0, showTotal = 0, winBase = 1, placeBase = 1, showBase = 1;
    poolTotalsArr.forEach(p => {
      let type = getStr(p, 'PoolType', '');
      if (type === 'WN') { winTotal = getVal(p, 'Amount', 0); winBase = getVal(p, 'Base', 1) || 1; }
      else if (type === 'PL') { placeTotal = getVal(p, 'Amount', 0); placeBase = getVal(p, 'Base', 1) || 1; }
      else if (type === 'SH') { showTotal = getVal(p, 'Amount', 0); showBase = getVal(p, 'Base', 1) || 1; }
    });

    let entries = (payload.WPSPools && payload.WPSPools.Entries) || [];

    // Detect mode per pool (Win/Place/Show could in principle each be toggled independently,
    // though in practice it's likely one page-wide toggle) using the raw values across all horses.
    let rawWin = entries.map(e => getVal(e, 'Win', NaN));
    let rawPlace = entries.map(e => getVal(e, 'Place', NaN));
    let rawShow = entries.map(e => getVal(e, 'Show', NaN));
    let winNorm = normalizePoolValues(rawWin, winTotal);
    let placeNorm = normalizePoolValues(rawPlace, placeTotal);
    let showNorm = normalizePoolValues(rawShow, showTotal);

    let poolsMap = {};
    let anyShow = false;
    entries.forEach((e, i) => {
      let prog = getStr(e, 'ProgramNumber');
      let winAmt = winNorm.values[i];
      let placeAmt = placeNorm.values[i];
      let showAmt = showNorm.values[i];
      if (showAmt > 0) anyShow = true;
      poolsMap[prog] = {
        win: (isNaN(winAmt) || winAmt < 0) ? null : winAmt,
        place: (isNaN(placeAmt) || placeAmt < 0) ? null : placeAmt,
        show: (isNaN(showAmt) || showAmt <= 0) ? null : showAmt
      };
    });

    return {
      poolsMap, winTotal, placeTotal, showTotal, showAvailable: anyShow,
      detectedPercentMode: winNorm.wasPercent || placeNorm.wasPercent || showNorm.wasPercent
    };
  }

  // Real total return per $1 WIN bet (includes stake), via the standard pari-mutuel formula:
  // (Pool Total x (1 - takeout)) / $ wagered on this horse. Returns null if pool data unavailable.
  function getWinPoolPayout(program, stake) {
    let entry = cachedPoolsMap[String(program)];
    if (!entry || entry.win === null || entry.win <= 0 || !cachedWinPoolTotal) return null;
    let s = stake > 0 ? stake : 0; // the bettor's own stake joins this horse's money and the pool
    let netPool = (cachedWinPoolTotal + s) * (1 - getEffectiveTakeout());
    return netPool / (entry.win + s);
  }

  // Real total return per $1 PLACE bet — the CORRECT pari-mutuel mechanism, not the WIN formula.
  // Place/show pools are shared among however many horses actually finish in those positions: each
  // placer's own stake is returned as principal first, then the remaining PROFIT is split EQUALLY
  // among the placing horses (not proportionally to their stakes), and each horse's equal profit
  // share is divided by THAT horse's own stake — which is exactly why a heavily-bet favorite pays
  // less than a lightly-bet longshot even when both place. Verified against a real race: computed
  // this way, estimates landed within a few cents of the actual posted payouts for all 3 SHOW
  // horses; the old single-horse-takes-the-whole-pool formula overestimated by roughly 3x.
  //
  // Since we can't know in advance who ELSE will place, `assumedCoPlacerPrograms` should be the
  // model's own top-ranked contenders (excluding `program` itself) as the best available guess —
  // this is inherently an estimate, clearly labeled as such in the candidate's detail text.
  function getPlacePoolPayout(program, assumedCoPlacerPrograms, stake) {
    if (!cachedPlacePoolTotal) return null;
    let allPlacers = [String(program)].concat((assumedCoPlacerPrograms || []).map(String));
    let stakes = allPlacers.map(p => {
      let entry = cachedPoolsMap[p];
      return (entry && entry.place !== null && entry.place > 0) ? entry.place : null;
    });
    if (stakes.some(s => s === null)) return null;
    let s = stake > 0 ? stake : 0; // the bettor's own stake joins this horse's money and the pool
    let targetStake = stakes[0] + s;
    let combinedStakes = stakes.reduce((a, b) => a + b, 0) + s;
    let netPool = (cachedPlacePoolTotal + s) * (1 - getEffectiveTakeout());
    let profitPool = netPool - combinedStakes;
    if (profitPool <= 0) return null;
    let profitPerPlacer = profitPool / allPlacers.length;
    return 1 + (profitPerPlacer / targetStake);
  }

  // Real total return per $1 SHOW bet — same corrected mechanism as PLACE above, split among 3
  // assumed co-placers instead of 2.
  function getShowPoolPayout(program, assumedCoPlacerPrograms, stake) {
    if (!cachedShowPoolTotal) return null;
    let allPlacers = [String(program)].concat((assumedCoPlacerPrograms || []).map(String));
    let stakes = allPlacers.map(p => {
      let entry = cachedPoolsMap[p];
      return (entry && entry.show !== null && entry.show > 0) ? entry.show : null;
    });
    if (stakes.some(s => s === null)) return null;
    let s = stake > 0 ? stake : 0; // the bettor's own stake joins this horse's money and the pool
    let targetStake = stakes[0] + s;
    let combinedStakes = stakes.reduce((a, b) => a + b, 0) + s;
    let netPool = (cachedShowPoolTotal + s) * (1 - getEffectiveTakeout());
    let profitPool = netPool - combinedStakes;
    if (profitPool <= 0) return null;
    let profitPerPlacer = profitPool / allPlacers.length;
    return 1 + (profitPerPlacer / targetStake);
  }

  function getFormattedHorsesCsv() {
    let keys = Object.keys(cachedHorsesMap);
    if (keys.length === 0) return "Awaiting Horse Data...";
    let sorted = keys.map(k => cachedHorsesMap[k]).sort((a, b) => {
      let numA = parseInt(String(a.PROGRAM).replace(/\D/g, ''), 10) || 0;
      let numB = parseInt(String(b.PROGRAM).replace(/\D/g, ''), 10) || 0;
      return numA - numB;
    });

    let liveProgramNumbers = sorted.filter(h => !h.IS_SCRATCHED && !cachedScratchedMap[h.PROGRAM]).map(h => h.PROGRAM);

    // Extends each raw horse row with the DERIVED values the model actually computes/uses from
    // that raw data (parsed odds, real pool payout rates, one sample exacta combo) — so blanks or
    // parsing failures downstream of the raw fields are visible here too, not just missing raw data.
    let extendedRows = sorted.map(h => {
      let mlDec = parseOddsToDecimal(h.ML_ODDS);
      let liveDec = parseOddsToDecimal(h.LIVE_ODDS);
      let oddsNumeric = !isNaN(liveDec) && liveDec > 0 ? liveDec : mlDec;

      let poolEntry = cachedPoolsMap[h.PROGRAM];
      let winPayout = getWinPoolPayout(h.PROGRAM);
      let placePayout = getPlacePoolPayout(h.PROGRAM);
      let showPayout = getShowPoolPayout(h.PROGRAM);

      // One sample exacta combo per horse (paired with the first other live horse) — just enough
      // to confirm the Probables grid actually loaded data for this horse, without dumping the
      // full N-1 row of combos per horse.
      let otherProg = liveProgramNumbers.find(p => p !== h.PROGRAM);
      let exactaSample = '""';
      if (otherProg) {
        let payout = getExactaPayout(h.PROGRAM, otherProg);
        exactaSample = `"#${h.PROGRAM}/#${otherProg}: ${payout !== null ? '$' + payout.toFixed(2) : 'N/A'}"`;
      }

      return Object.assign({}, h, {
        ODDS_NUMERIC: !isNaN(oddsNumeric) ? oddsNumeric : '',
        WIN_POOL_AMT: (poolEntry && poolEntry.win !== null) ? poolEntry.win : '',
        PLACE_POOL_AMT: (poolEntry && poolEntry.place !== null) ? poolEntry.place : '',
        SHOW_POOL_AMT: (poolEntry && poolEntry.show !== null) ? poolEntry.show : '',
        WIN_PAYOUT_RATE: winPayout !== null ? winPayout.toFixed(2) : '',
        PLACE_PAYOUT_RATE: placePayout !== null ? placePayout.toFixed(2) : '',
        SHOW_PAYOUT_RATE: showPayout !== null ? showPayout.toFixed(2) : '',
        EXACTA_SAMPLE: exactaSample
      });
    });

    let headerRow = Object.keys(extendedRows[0]).join(',');
    let dataRows = extendedRows.map(h => Object.values(h).join(',')).join('\n');
    return `${headerRow}\n${dataRows}`;
  }

    // Tracks using the back-tested "Special" weight set instead of default Thoroughbred weights.
    // Deliberately hardcoded rather than inferred from live pool size — pool totals can nearly
    // double in the final minute to post at major tracks (computer-automated betting), which would
    // make a live-data threshold change the weight set out from under a bet you already decided on.
    // Only add a track here after independently testing it the way Churchill and Horseshoe
    // Indianapolis were tested — this list should never be a guess.
    const SPECIAL_WEIGHT_TRACKS = ['churchill', 'horseshoe indianapolis'];

    let isSpecialTrack = cleanBreed === 'Thoroughbred' &&
      SPECIAL_WEIGHT_TRACKS.some(t => cleanTrack.toLowerCase().includes(t));

    let activeWeightKey = forcedWeightKey || (isSpecialTrack ? 'Special' : (weightsDict[cleanBreed] ? cleanBreed : 'Thoroughbred'));
    let activeWeights = Object.assign({}, weightsDict[activeWeightKey]);

    let weekRaces = 0, meetRaces = 100;
    let weekBias = 0, meetBias = 0;
    let weekIV = { RAIL: 1.0, "1-3": 1.0, "4-7": 1.0, "8+": 1.0, E: 1.0, EP: 1.0, P: 1.0, S: 1.0 };
    let meetIV = { RAIL: 1.0, "1-3": 1.0, "4-7": 1.0, "8+": 1.0, E: 1.0, EP: 1.0, P: 1.0, S: 1.0 };

    if (cachedStatsObj) {
      let s = cachedStatsObj;
      weekRaces = getVal(s, ['weekCountRaces', 'weekRaces'], 0);
      meetRaces = getVal(s, ['meetCountRaces', 'meetRaces', 'meetCount'], 100);

      function normB(v) { let n = parseFloat(v) || 0; return n > 1.0 ? n / 100.0 : n; }
      weekBias = normB(getVal(s, 'weekSpeedBias'));
      meetBias = normB(getVal(s, 'meetSpeedBias'));

      weekIV = {
        RAIL: getVal(s, 'weekPost1Impact', 1.0),
        "1-3": getVal(s, 'weekPost2Impact', 1.0),
        "4-7": getVal(s, 'weekPost3Impact', 1.0),
        "8+": getVal(s, 'weekPost4Impact', 1.0),
        E: getVal(s, 'weekRs1Impact', 1.0),
        EP: getVal(s, 'weekRs2Impact', 1.0),
        P: getVal(s, 'weekRs3Impact', 1.0),
        S: getVal(s, 'weekRs4Impact', 1.0)
      };

      meetIV = {
        RAIL: getVal(s, 'meetPost1Impact', 1.0),
        "1-3": getVal(s, 'meetPost2Impact', 1.0),
        "4-7": getVal(s, 'meetPost3Impact', 1.0),
        "8+": getVal(s, 'meetPost4Impact', 1.0),
        E: getVal(s, 'meetRs1Impact', 1.0),
        EP: getVal(s, 'meetRs2Impact', 1.0),
        P: getVal(s, 'meetRs3Impact', 1.0),
        S: getVal(s, 'meetRs4Impact', 1.0)
      };
    }

    let wWeek = 0.0, wMeet = 1.0;
    let ruleAppliedMsg = "";

    if (weekRaces >= 15) {
      wWeek = 0.65; wMeet = 0.35;
      ruleAppliedMsg = `[Rule Applied] Blending 65% Week / 35% Meet (Week sample sufficient: ${Math.round(weekRaces)} >= 15 races)`;
    } else if (weekRaces >= 7) {
      wWeek = 0.20; wMeet = 0.80;
      ruleAppliedMsg = `[Rule Applied] Blending 20% Week / 80% Meet (${Math.round(weekRaces)} week races)`;
    } else {
      wWeek = 0.00; wMeet = 1.00;
      ruleAppliedMsg = `[Rule Applied] Using 100% Meet stats (Week sample too small: ${Math.round(weekRaces)} < 7 races)`;
    }

    ruleAppliedMsg += ` | [Weight Set: ${activeWeightKey}]`;

    let wetnessTier = getWetnessTier(cachedTrackCondition);
    let isWet = wetnessTier > 0;
    if (isWet) {
      let tierLabel = wetnessTier === 2 ? 'Tier 2: Sloppy/Muddy/Heavy' : 'Tier 1: Off/Yielding';
      ruleAppliedMsg += ` | [Track Condition: ${cachedTrackCondition.toUpperCase()} — ${tierLabel} — Power Bonus Zeroed]`;
    }

    let activeSpeedBias = (wWeek * weekBias) + (wMeet * meetBias);
    let postIvMapping = {
      "RAIL": Math.round(((wWeek * weekIV.RAIL) + (wMeet * meetIV.RAIL)) * 100) / 100,
      "1-3": Math.round(((wWeek * weekIV["1-3"]) + (wMeet * meetIV["1-3"])) * 100) / 100,
      "4-7": Math.round(((wWeek * weekIV["4-7"]) + (wMeet * meetIV["4-7"])) * 100) / 100,
      "8+": Math.round(((wWeek * weekIV["8+"]) + (wMeet * meetIV["8+"])) * 100) / 100
    };

    let runStyleIvMapping = {
      "E": Math.round(((wWeek * weekIV.E) + (wMeet * meetIV.E)) * 100) / 100,
      "E/P": Math.round(((wWeek * weekIV.EP) + (wMeet * meetIV.EP)) * 100) / 100,
      "P": Math.round(((wWeek * weekIV.P) + (wMeet * meetIV.P)) * 100) / 100,
      "S": Math.round(((wWeek * weekIV.S) + (wMeet * meetIV.S)) * 100) / 100
    };

    let activeMeetCount = meetRaces;
    let sampleScaleFactor = 0.0;
    if (activeMeetCount < 6) sampleScaleFactor = 0.0;
    else if (activeMeetCount <= 15) sampleScaleFactor = 0.10;
    else sampleScaleFactor = 1.00;

    let biasDecimal = activeSpeedBias > 1.0 ? activeSpeedBias / 100.0 : activeSpeedBias;
    if (biasDecimal > 0.55) {
      let rawEarlyBoost = (biasDecimal - 0.55) * 0.5;
      activeWeights.W_Early += (rawEarlyBoost * sampleScaleFactor);
    }

    let postIvMultiplier = 6.0, styleIvMultiplier = 6.0;
    if (cleanBreed === 'Harness') { postIvMultiplier = 7.0; styleIvMultiplier = 7.0; }
    else if (cleanBreed === 'Quarter Horse') { postIvMultiplier = 4.0; styleIvMultiplier = 5.0; }

    function getPostCat(postVal) {
      let p = parseInt(postVal, 10);
      if (p === 1) return "RAIL";
      if (p >= 1 && p <= 3) return "1-3";
      if (p >= 4 && p <= 7) return "4-7";
      if (p >= 8) return "8+";
      return "1-3";
    }

    function calcPostBonus(cat) {
      if (activeMeetCount < 15) return 0.0;
      let iv = postIvMapping[cat] || 1.0;
      let diff = Math.max(0.0, iv - 1.0);
      return diff >= 0.03 ? Math.round(Math.min(5.0, diff * postIvMultiplier) * 100) / 100 : 0.0;
    }

    function calcStyleBonus(code, pts) {
      if (activeMeetCount < 15) return 0.0;
      let iv = runStyleIvMapping[code] || 1.0;
      let diff = Math.max(0.0, iv - 1.0);
      if (diff >= 0.03) {
        let ptsFactor = Math.min(Math.max(parseFloat(pts) / 8.0, 0.0), 1.0);
        return Math.round(Math.min(5.0, diff * styleIvMultiplier * ptsFactor) * 100) / 100;
      }
      return 0.0;
    }

    // Wet-track early-speed bonus: on off/sloppy/muddy tracks, front-runners and early-pace
    // horses tend to benefit (clear of kickback, firmer worked-up ground). Pure closers get none.
    // This stacks ON TOP OF calcStyleBonus's 5.0 cap rather than sharing it, since it reflects a
    // separate signal (today's track condition) rather than the meet/week run-style IV data.
    const WET_STYLE_FACTOR = { 'E': 1.0, 'E/P': 0.65, 'P': 0.25, 'S': 0.0 };
    const WET_TIER_MAX_BONUS = { 1: 0.75, 2: 1.5 }; // tier 1 = off/yielding, tier 2 = sloppy/muddy/heavy

    function calcWetStyleBonus(code, pts, tier) {
      if (!tier || tier < 1) return 0.0;
      let styleFactor = WET_STYLE_FACTOR[code] !== undefined ? WET_STYLE_FACTOR[code] : 0.0;
      if (styleFactor <= 0.0) return 0.0;
      let ptsFactor = Math.min(Math.max(parseFloat(pts) / 8.0, 0.0), 1.0);
      let maxBonus = WET_TIER_MAX_BONUS[tier] || 0.0;
      return Math.round(maxBonus * styleFactor * ptsFactor * 100) / 100;
    }

    let runners = Object.values(cachedHorsesMap).map(h => Object.assign({}, h));

    let numCols = SCORING_NUM_FIELDS;
    numCols.forEach(col => {
      let validVals = runners.map(r => parseFloat(r[col]) || 0).filter(v => v > 0);
      let avg = validVals.length > 0 ? validVals.reduce((a, b) => a + b, 0) / validVals.length : 0.0;
      runners.forEach(r => {
        let v = parseFloat(r[col]) || 0;
        if (v <= 0) {
          r[col] = Math.round(avg * 10) / 10;
          r.IMPUTED_COUNT = (r.IMPUTED_COUNT || 0) + 1; // how many scoring fields were filled in (Research Export)
        }
      });
    });

    let pwrVals = runners.map(r => parseFloat(r.PRM_PWR) || 0).sort((a, b) => a - b);
    let medianPwr = 0;
    if (pwrVals.length > 0) {
      let mid = Math.floor(pwrVals.length / 2);
      medianPwr = pwrVals.length % 2 !== 0 ? pwrVals[mid] : (pwrVals[mid - 1] + pwrVals[mid]) / 2;
    }

    runners.forEach(r => {
      let mlDec = parseOddsToDecimal(r.ML_ODDS);
      let liveDec = parseOddsToDecimal(r.LIVE_ODDS);
      let finalOddsDec = !isNaN(liveDec) && liveDec > 0 ? liveDec : mlDec;
      r.ODDS_NUMERIC = finalOddsDec;

      let avgDist = parseFloat(r.AVG_DIST_SPD) || 0;
      let spdLr = parseFloat(r.SPD_LR) || 0;
      let avgSpd = parseFloat(r.AVG_SPD) || 0;
      let cSpeed = 0;
      if (avgDist > 0 && spdLr > 0) cSpeed = (0.70 * avgDist) + (0.30 * spdLr);
      else if (spdLr > 0) cSpeed = spdLr;
      else if (avgDist > 0) cSpeed = avgDist;
      else cSpeed = avgSpd;

      let rawPower = parseFloat(r.PRM_PWR) || 0;
      let cPower = rawPower;

      let powerBonus = (isWet || rawPower <= medianPwr) ? 0.0 : 0.0; 

      let avgCls = parseFloat(r.AVG_CLS) || 0;
      let lastCls = parseFloat(r.LAST_CLS) || 0;
      let cClass = avgCls > 0 ? avgCls : (lastCls > 0 ? lastCls : 0);


      // Win rates arrive as fractions (0.14), but every other component is on a ~25-160 scale, so they are converted to
      // percentage points (14) here; otherwise the driver/trainer weights would have almost no effect on the score.
      // A value already above 1 is taken to be in percent and left alone.
      let jkyRaw = parseFloat(r.W_JKY) || 0;
      let trnRaw = parseFloat(r.W_TRN) || 0;
      let cDriver = jkyRaw > 1 ? jkyRaw : jkyRaw * 100;
      let cTrainer = trnRaw > 1 ? trnRaw : trnRaw * 100;

      let e1 = parseFloat(r.E1) || 0;
      let e2 = parseFloat(r.E2) || 0;
      let cEarly = (e1 > 0 && e2 > 0) ? (0.50 * e1 + 0.50 * e2) : (e2 > 0 ? e2 : e1);

      let cFinish = parseFloat(r.LP) || 0;

      let days = parseFloat(r.DAYS_OFF) || 0;
      let cRecency = 75.0;
      if (days >= 14 && days <= 45) cRecency = 100.0;
      else if (days < 14) cRecency = 85.0;
      else if (days >= 46 && days <= 90) cRecency = 70.0;
      else if (days > 90) cRecency = 50.0;

      let cMarket = !isNaN(finalOddsDec) && finalOddsDec > 0 ? (1.0 / (finalOddsDec + 1.0)) * 100.0 : 0.0;

      let baseSkill = 
        (cSpeed * activeWeights.W_Speed) +
        (cPower * activeWeights.W_Power) +
        (cClass * activeWeights.W_Class) +
        (cDriver * activeWeights.W_Driver) +
        (cTrainer * activeWeights.W_Trainer) +
        (cEarly * activeWeights.W_Early) +
        (cFinish * activeWeights.W_Finish) +
        (cRecency * activeWeights.W_Recency) +
        (cMarket * activeWeights.W_Market);

      let postCat = getPostCat(r.POST);
      let postBonus = calcPostBonus(postCat);
      let styleBonus = calcStyleBonus(r.RUN_STYLE, r.RUN_STYLE_PTS);
      let wetBonus = calcWetStyleBonus(r.RUN_STYLE, r.RUN_STYLE_PTS, wetnessTier);

      // Component values kept on the runner so the Research Export can log what the weights act on.
      r.C_SPEED = cSpeed; r.C_POWER = cPower; r.C_CLASS = cClass; r.C_DRIVER = cDriver;
      r.C_TRAINER = cTrainer; r.C_EARLY = cEarly; r.C_FINISH = cFinish; r.C_RECENCY = cRecency; r.C_MARKET = cMarket;

      r.POWER_BONUS = powerBonus;
      r.BASE_SKILL = Math.round(baseSkill * 100) / 100;
      r.POST_BONUS = postBonus;
      r.STYLE_BONUS = styleBonus;
      r.WET_BONUS = wetBonus;

      // Calculate candidate final score
      let calculatedScore = Math.round((baseSkill + postBonus + styleBonus + powerBonus + wetBonus) * 100) / 100;

      // Zero out final score at the very end if scratched
      if (r.IS_SCRATCHED === true || cachedScratchedMap[r.PROGRAM]) {
        r.FINAL_SCORE = 0;
      } else {
        r.FINAL_SCORE = calculatedScore;
      }
    });

    runners.sort((a, b) => b.FINAL_SCORE - a.FINAL_SCORE);
    runners.forEach((r, idx) => { r.RANK = idx + 1; });

    let topField = runners.slice(0, 5);
    let top1 = topField[0];
    let top2 = topField[1];
    let top3 = topField[2];
    let top4 = topField[3];
    let top5 = topField[4];

    let s1 = top1 ? top1.FINAL_SCORE : 0;
    let s2 = top2 ? top2.FINAL_SCORE : 0;
    let s3 = top3 ? top3.FINAL_SCORE : 0;
    let s4 = top4 ? top4.FINAL_SCORE : 0;
    let s5 = top5 ? top5.FINAL_SCORE : 0;

    let gap12 = Math.round((s1 - s2) * 100) / 100;
    let gap23 = Math.round((s2 - s3) * 100) / 100;
    let gap34 = Math.round((s3 - s4) * 100) / 100;
    let gap45 = Math.round((s4 - s5) * 100) / 100;

    let gap13 = Math.round((s1 - s3) * 100) / 100;
    let gap14 = Math.round((s1 - s4) * 100) / 100;

    // ---------------------------------------------------------------
    // WAGER RECOMMENDATION ENGINE (probability + expected-value based)
    // ---------------------------------------------------------------
    // Replaces the old static point-gap tiers with:
    //  1) A blended win-probability model (market-implied probability, tilted by our score edge)
    //  2) Harville-derived exacta/box hit probabilities from those win probabilities
    //  3) Real exacta payouts from the Probables tab (cachedExactaGrid) when available,
    //     falling back to a generalized odds-derived estimate only if that grid hasn't loaded yet
    //  4) Expected value (EV) per dollar for every candidate bet, dynamically priced $2-$6
    //     off the size of the edge, instead of fixed tier dollar amounts

    let liveRunners = runners.filter(r => !(r.IS_SCRATCHED === true || cachedScratchedMap[r.PROGRAM]));
    let winProbs = {}; // program -> { modelProb, zScore }
    const MAX_Z_REF = 2.5; // cap on standardized score, avoids runaway probabilities/conviction from outlier scores

    // A "typical" leader in a modest field sits roughly 1-1.5 SD above the field average purely
    // from order-statistics (the best of N horses is expected to look somewhat above-average even
    // with no real edge) — a plain z/MAX_Z_REF mapping was treating that as meaningfully confident,
    // which structurally skewed stakes toward $4 for perfectly ordinary favorites. BASELINE_Z shifts
    // the zero-point: only a leader clearly ABOVE typical starts scaling stake past the $2 floor.
    const BASELINE_Z = 1.3;
    function computeConviction(zScore) {
      let excess = Math.max(0, zScore - BASELINE_Z);
      return Math.min(excess / (MAX_Z_REF - BASELINE_Z), 1);
    }

    // Win probability is derived ONLY from the model's own FINAL_SCORE (which already has its own
    // market/odds component baked in via W_Market — see baseSkill). Live odds are NOT blended in here;
    // they're used later strictly to price the WIN bet's payout, per the design correction below.
    if (liveRunners.length >= 2) {
      let scoreVals = liveRunners.map(r => r.FINAL_SCORE);
      let meanScore = scoreVals.reduce((a, b) => a + b, 0) / scoreVals.length;
      let variance = scoreVals.reduce((a, b) => a + Math.pow(b - meanScore, 2), 0) / scoreVals.length;
      let stdevScore = Math.sqrt(variance) || 1;

      const SCORE_PROB_K = 0.70; // controls how sharply score gaps translate into probability separation
      // Every hit % is built from these win probabilities (Harville), so one blend toward an even split makes all of
      // them less confident at once: the model can't see most of what decides a race, so a leader's raw share overstates
      // its real chance. 0 = raw model, 1 = every horse equal. Near-tie clustering below keeps using the RAW share, so
      // which bets are eligible does not change; only the probabilities, EVs and EV-based picks do.
      const PROB_SHRINK = 0.15;

      let zArr = liveRunners.map(r => Math.max(-MAX_Z_REF, Math.min(MAX_Z_REF, (r.FINAL_SCORE - meanScore) / stdevScore)));
      let rawArr = zArr.map(z => Math.exp(SCORE_PROB_K * z));
      let rawSum = rawArr.reduce((a, b) => a + b, 0) || 1;

      liveRunners.forEach((r, i) => {
        let rawShare = rawArr[i] / rawSum;
        let shrunk = (1 - PROB_SHRINK) * rawShare + PROB_SHRINK / liveRunners.length;
        winProbs[r.PROGRAM] = { modelProb: shrunk, rawProb: rawShare, zScore: zArr[i] };
        r.WIN_PROB = shrunk; r.Z_SCORE = zArr[i]; // exposed for the Research Export (WIN_PROB is the shrunk value every bet uses)
      });
    }

    // Harville extension: P(A 1st, B 2nd) = P(A) * P(B) / (1 - P(A))
    function exactaOrderProb(progA, progB) {
      let pA = winProbs[progA] ? winProbs[progA].modelProb : 0;
      let pB = winProbs[progB] ? winProbs[progB].modelProb : 0;
      if (pA <= 0 || pA >= 1) return 0;
      return pA * (pB / (1 - pA));
    }

    // Probability horse finishes 1st OR 2nd (top-2 / "place"), purely from our own score-based
    // winProbs — same Harville logic, summed over every other horse's chance of winning with
    // this one taking 2nd behind them. No market data involved, consistent with the WIN model.
    function placeProbabilityOf(prog) {
      let pi = winProbs[prog] ? winProbs[prog].modelProb : 0;
      if (pi <= 0) return 0;
      let sumSecond = 0;
      liveRunners.forEach(r => {
        if (String(r.PROGRAM) === String(prog)) return;
        let pj = winProbs[r.PROGRAM] ? winProbs[r.PROGRAM].modelProb : 0;
        if (pj >= 1) return;
        sumSecond += pj * (pi / (1 - pj));
      });
      return Math.min(1, pi + sumSecond);
    }

    // Probability horse finishes 1st, 2nd, OR 3rd ("show"), extending the same Harville logic one
    // level further: sum over every ordered pair of other horses taking 1st and 2nd, times this
    // horse taking 3rd given both are gone. Purely score-derived, no market data — consistent with
    // WIN and PLACE above.
    function showProbabilityOf(prog) {
      let pi = winProbs[prog] ? winProbs[prog].modelProb : 0;
      if (pi <= 0) return 0;
      let base = placeProbabilityOf(prog); // covers 1st + 2nd
      let thirdSum = 0;
      liveRunners.forEach(rj => {
        if (String(rj.PROGRAM) === String(prog)) return;
        let pj = winProbs[rj.PROGRAM] ? winProbs[rj.PROGRAM].modelProb : 0;
        if (pj <= 0 || pj >= 1) return;
        liveRunners.forEach(rk => {
          if (String(rk.PROGRAM) === String(prog) || String(rk.PROGRAM) === String(rj.PROGRAM)) return;
          let pk = winProbs[rk.PROGRAM] ? winProbs[rk.PROGRAM].modelProb : 0;
          let denom2 = 1 - pj;
          if (denom2 <= 0) return;
          let pkGivenJ = pk / denom2;
          let denom3 = 1 - pj - pk;
          if (denom3 <= 0) return;
          thirdSum += pj * pkGivenJ * (pi / denom3);
        });
      });
      return Math.min(1, base + thirdSum);
    }

    // Real live probable payout ONLY — no odds-based fallback. If the real grid doesn't have this
    // combo loaded yet, returns null so the candidate simply isn't generated until real data is
    // available, matching how PLACE and SHOW already behave. The old fallback (multiplying both
    // horses' odds together) was a reasonable rough approximation for one combo, but broke down
    // badly once summed across all 6 combos of a 3-horse box simultaneously — waiting for real data
    // is more honest than bounding a formula that shouldn't be trusted at all.
    function exactaPayoutEstimate(progA, progB, unitStake, ticketTotal) {
      let real = getExactaPayout(progA, progB, unitStake, ticketTotal);
      return real !== null ? { value: real, isReal: true } : null;
    }

    // Harville extension one level further: P(A 1st, B 2nd, C 3rd) = P(A) * [P(B)/(1-P(A))] *
    // [P(C)/(1-P(A)-P(B))]. Purely score-derived, consistent with every other probability here.
    function trifectaOrderProb(progA, progB, progC) {
      let pA = winProbs[progA] ? winProbs[progA].modelProb : 0;
      let pB = winProbs[progB] ? winProbs[progB].modelProb : 0;
      let pC = winProbs[progC] ? winProbs[progC].modelProb : 0;
      if (pA <= 0 || pA >= 1) return 0;
      let denom2 = 1 - pA;
      if (denom2 <= 0) return 0;
      let pBgivenA = pB / denom2;
      let denom3 = 1 - pA - pB;
      if (denom3 <= 0) return 0;
      let pCgivenAB = pC / denom3;
      return pA * pBgivenA * pCgivenAB;
    }

    // There's no real trifecta payout matrix (unlike exacta's Probables grid) — only the pool's
    // TOTAL size, with no data on how it's actually split across combos. Estimating a combo's
    // dollar share purely from our own modeled probability would be circular (the same number
    // would price the bet AND judge whether it's worth betting, so it could never show a real
    // edge). Instead: anchor on the REAL exacta payout for the top two legs (genuine market money),
    // then scale by the model's own conditional probability of the third leg — grounds 2/3 of the
    // estimate in real data rather than being 100% self-referential. The trifecta pool TOTAL sets a
    // race-specific sanity ceiling: the most any single combo could plausibly pay is the whole net
    // pool divided by the smallest realistic wager on it (one base unit) — this scales correctly
    // with how big or small THIS race's trifecta pool actually is, rather than one flat number
    // applied to every race regardless of size.
    const TRIFECTA_PAYOUT_ABSOLUTE_BACKSTOP = 1000; // per $1 base; a hard backstop only, in case pool data is itself glitched
    function trifectaPayoutEstimate(progA, progB, progC, unitStake, ticketTotal) {
      let exactaLeg = getExactaPayout(progA, progB);
      if (exactaLeg === null) return null; // no real 2-leg anchor available yet
      let pA = winProbs[progA] ? winProbs[progA].modelProb : 0;
      let pB = winProbs[progB] ? winProbs[progB].modelProb : 0;
      let pC = winProbs[progC] ? winProbs[progC].modelProb : 0;
      let denom3 = 1 - pA - pB;
      if (denom3 <= 0) return null;
      let pCgivenAB = pC / denom3;
      if (pCgivenAB <= 0) return null;
      let est = exactaLeg / pCgivenAB;

      let netTriPool = cachedTrifectaPoolTotal * (1 - TAKEOUT_RATE);
      let raceSpecificCeiling = cachedTrifectaPoolTotal > 0
        ? netTriPool / TRIFECTA_MIN_BASE_UNIT
        : TRIFECTA_PAYOUT_ABSOLUTE_BACKSTOP;

      let capped = Math.min(est, raceSpecificCeiling, TRIFECTA_PAYOUT_ABSOLUTE_BACKSTOP);
      // The bettor's own money lowers the payout the same way it does for any pool; the trifecta pool total is only known
      // when the Probables tab has loaded, otherwise this is a no-op.
      return adjustPayoutForOwnStake(capped, cachedTrifectaPoolTotal, TAKEOUT_RATE, unitStake, ticketTotal);
    }

    // Dynamic $2-$6 stake sizing off an edge fraction (0-1), rounded to the nearest stepUnit
    function scaleStake(edgeFraction, minTotal, maxTotal, stepUnit) {
      let clamped = Math.max(0, Math.min(1, edgeFraction));
      let raw = minTotal + clamped * (maxTotal - minTotal);
      let stepped = Math.round(raw / stepUnit) * stepUnit;
      return Math.max(minTotal, Math.min(maxTotal, stepped));
    }

    let candidates = [];
    let customWagerResult = null; // the user's custom wager, priced at the end of the engine block below
    let baseU = cachedExactaBaseUnit || 1;

    if (liveRunners.length >= 2) {
      let ranked = liveRunners.slice().sort((a, b) =>
        (winProbs[b.PROGRAM] ? winProbs[b.PROGRAM].modelProb : 0) - (winProbs[a.PROGRAM] ? winProbs[a.PROGRAM].modelProb : 0)
      );
      let c1 = ranked[0], c2 = ranked[1], c3 = ranked[2], c4 = ranked[3], c5 = ranked[4];

      // Uses win-probability ratio, not a raw score-point gap, so it scales correctly whether the
      // field is tightly bunched or widely spread (same reasoning as the rest of this engine).
      // Lowered from 0.85 -> 0.75: real logged results showed 0.85 was too loose to catch genuinely
      // close situations (several exacta-box rows had a tiny point gap to the excluded horse that
      // never triggered near-tie reclassification) — 0.75 better reflects how unpredictable racing
      // actually is.
      const NEAR_TIE_PROB_RATIO = 0.75;

      function isNearTie(progA, progB) {
        let pA = winProbs[progA] ? winProbs[progA].rawProb : 0;
        let pB = winProbs[progB] ? winProbs[progB].rawProb : 0;
        let hi = Math.max(pA, pB), lo = Math.min(pA, pB);
        if (hi <= 0) return false;
        return (lo / hi) >= NEAR_TIE_PROB_RATIO;
      }

      let c1c2Tie = !!(c1 && c2 && isNearTie(c1.PROGRAM, c2.PROGRAM));
      let c2c3Tie = !!(c2 && c3 && isNearTie(c2.PROGRAM, c3.PROGRAM));
      let c3c4Tie = !!(c3 && c4 && isNearTie(c3.PROGRAM, c4.PROGRAM));
      let c4c5Tie = !!(c4 && c5 && isNearTie(c4.PROGRAM, c5.PROGRAM));

      // BOUNDARY RULE (applies to every bet for Option A/B eligibility): the first horse NOT covered
      // by a bet must not be a near-tie with the weakest horse the bet does cover — otherwise the
      // bet leaves out a horse that is just as likely as the ones it includes. Per bet:
      //   Win: c2 vs c1 | Place: c3 vs the horse | Show: c4 vs the horse | Straight: c3 vs c2 |
      //   Box(2): c3 vs c2 | Box(3): c4 vs c3 | Wheel(3): c4 vs c3 | Wheel(4): c5 vs c4 |
      //   Trifecta Box(3): c4 vs c3 | Trifecta Box(4): c5 vs c4

      // --- WIN on top model pick — defensible only when c1 is genuinely more likely than c2, not
      // just "above the field average." ---
      // Conviction (and therefore stake size) comes ONLY from how much of a score outlier the top
      // pick is (its z-score) — never from odds or pool data.
      if (c1) {
        let winConviction = computeConviction(winProbs[c1.PROGRAM].zScore);
        let stake = scaleStake(Math.min(winConviction, 1), 2, 6, 1);
        let realWinPayoutRate = getWinPoolPayout(c1.PROGRAM, stake); // priced with this stake already in the pool
        if (realWinPayoutRate !== null) {
          let pWin = winProbs[c1.PROGRAM].modelProb;
          let payoutIfWin = stake * realWinPayoutRate;
          let ev = pWin * payoutIfWin - stake;
          candidates.push({
            type: 'WIN', typeLabel: 'Win', horses: `#${c1.PROGRAM}`,
            label: `$${stake} WIN on #${c1.PROGRAM}`,
            cost: stake, hitProb: pWin, ev, evPerDollar: ev / stake, eligible: !c1c2Tie,
            payoutLow: payoutIfWin, payoutHigh: payoutIfWin,
            isRealData: true,
            detail: `Model Win Prob ${(pWin * 100).toFixed(0)}% (Live Pool Payout)`
          });
        }
      }

      // --- PLACE: evaluates BOTH c1 and c2 independently (not just "always the model's #1 pick")
      // — each candidate's own boundary threat is c3 (the horse that could bump it out of the top
      // two), so each needs its own separation-from-c3 check to be trustworthy. Both can be
      // generated at once; the EV/hit-rate selection naturally picks whichever is actually better,
      // rather than the engine hard-coding which horse "deserves" the bet.
      // Place/Show payouts depend on WHO ELSE places (each placer's stake comes out of the shared
      // pool, so a heavily-bet co-placer drags everyone's payout down). We can't know that in
      // advance, so instead of trusting one guessed co-placer set we price EVERY plausible set drawn
      // from the model's top-5 contenders: payoutLow = worst case (what the 1.4x payout gate tests),
      // payoutHigh = best case, and EV uses the average across sets.
      let contenders = ranked.slice(0, 5);
      function summarizeRates(rates) {
        rates = rates.filter(r => r !== null);
        if (rates.length === 0) return null;
        return {
          min: Math.min(...rates), max: Math.max(...rates),
          avg: rates.reduce((x, y) => x + y, 0) / rates.length, n: rates.length
        };
      }
      function placeRateRange(horse, stake) {
        let others = contenders.filter(h => h.PROGRAM !== horse.PROGRAM);
        return summarizeRates(others.map(o => getPlacePoolPayout(horse.PROGRAM, [o.PROGRAM], stake)));
      }
      function showRateRange(horse, stake) {
        let others = contenders.filter(h => h.PROGRAM !== horse.PROGRAM);
        let rates = [];
        for (let i = 0; i < others.length; i++) {
          for (let j = i + 1; j < others.length; j++) {
            rates.push(getShowPoolPayout(horse.PROGRAM, [others[i].PROGRAM, others[j].PROGRAM], stake));
          }
        }
        return summarizeRates(rates);
      }

      function buildPlaceCandidate(horse) {
        if (!horse || !c3) return;
        let placeEligible = !isNearTie(horse.PROGRAM, c3.PROGRAM); // boundary threat from c3 (gates Option A/B eligibility only)
        let conviction = computeConviction(winProbs[horse.PROGRAM].zScore);
        let stake = scaleStake(Math.min(conviction, 1), 2, 6, 1);
        let range = placeRateRange(horse, stake); // priced with this stake already in the pool
        if (range === null) return;
        let pPlace = placeProbabilityOf(horse.PROGRAM);
        if (pPlace <= 0) return;
        let ev = pPlace * (stake * range.avg) - stake;
        candidates.push({
          type: 'PLACE', typeLabel: 'Place', horses: `#${horse.PROGRAM}`,
          label: `$${stake} PLACE on #${horse.PROGRAM}`,
          cost: stake, hitProb: pPlace, ev, evPerDollar: ev / stake, eligible: placeEligible,
          payoutLow: stake * range.min, payoutHigh: stake * range.max,
          isRealData: true,
          detail: `Model Place Prob ${(pPlace * 100).toFixed(0)}% (Live Pool Payout, range across ${range.n} possible co-placers; EV uses the average)`
        });
      }
      buildPlaceCandidate(c1);
      buildPlaceCandidate(c2);

      // --- SHOW: evaluates c1, c2, AND c3 independently — boundary threat for each is c4. ---
      function buildShowCandidate(horse) {
        if (!horse || !c4) return;
        let showEligible = !isNearTie(horse.PROGRAM, c4.PROGRAM);
        let conviction = computeConviction(winProbs[horse.PROGRAM].zScore);
        let stake = scaleStake(Math.min(conviction, 1), 2, 6, 1);
        let range = showRateRange(horse, stake); // priced with this stake already in the pool
        if (range === null) return;
        let pShow = showProbabilityOf(horse.PROGRAM);
        if (pShow <= 0) return;
        let ev = pShow * (stake * range.avg) - stake;
        candidates.push({
          type: 'SHOW', typeLabel: 'Show', horses: `#${horse.PROGRAM}`,
          label: `$${stake} SHOW on #${horse.PROGRAM}`,
          cost: stake, hitProb: pShow, ev, evPerDollar: ev / stake, eligible: showEligible,
          payoutLow: stake * range.min, payoutHigh: stake * range.max,
          isRealData: true,
          detail: `Model Show Prob ${(pShow * 100).toFixed(0)}% (Live Pool Payout, range across ${range.n} possible co-placer pairs; EV uses the average)`
        });
      }
      if (c1 && c2 && c3) {
        buildShowCandidate(c1);
        buildShowCandidate(c2);
        buildShowCandidate(c3);
      }

      // --- Exacta Straight: c1 over c2 — only when c1 is CLEARLY ahead of c2 (confident who wins)
      // AND c2 is CLEARLY ahead of c3 (confident who's second, not threatened by c3). ---
      if (c1 && c2) {
        let pOrder = exactaOrderProb(c1.PROGRAM, c2.PROGRAM);
        let payout = exactaPayoutEstimate(c1.PROGRAM, c2.PROGRAM);
        if (payout && payout.value > 0 && pOrder > 0) {
          let fairPayout = 1 / pOrder;
          let edge = Math.max(0, (payout.value - fairPayout) / fairPayout);
          let baseStake = Math.max(baseU, scaleStake(Math.min(edge, 1), 2, 6, baseU));
          let cost = baseStake * 1;
          // stake sizing above uses the live probable; the payout the ticket is judged on includes its own money in the pool
          let ownPay = exactaPayoutEstimate(c1.PROGRAM, c2.PROGRAM, cost, cost);
          let payoutIfHit = (ownPay ? ownPay.value : payout.value) * baseStake;
          let ev = pOrder * payoutIfHit - cost;
          candidates.push({
            type: 'EXACTA_STRAIGHT', typeLabel: 'Exacta (Straight)', horses: `#${c1.PROGRAM}/#${c2.PROGRAM}`,
            label: `$${cost.toFixed(0)} Straight Exacta: #${c1.PROGRAM} / #${c2.PROGRAM}`,
            cost, hitProb: pOrder, ev, evPerDollar: ev / cost, eligible: !c1c2Tie && !c2c3Tie,
            payoutLow: payoutIfHit * 0.85, payoutHigh: payoutIfHit * 1.15,
            isRealData: true,
            detail: `Live Probable $${payout.value.toFixed(2)}/$1 base`
          });
        }
      }

      // --- Exacta Wheel (3 and 4 horse): c1 over the next 2 (or 3) horses, ONE direction only.
      // Both sizes are always generated so the expanded table can show them; `eligible` marks
      // whether the original clustering rule would have picked it for Option A/B: c1 CLEARLY
      // ahead (!c1c2Tie) with tied chasers (c2c3Tie), sized to 3 wheeled horses only if c4 is also
      // tied to c3. ---
      if (c1 && c2 && c3) {
        [2, 3].forEach(nLegs => {
          let wheelHorses = [c2, c3];
          if (nLegs === 3) { if (!c4) return; wheelHorses.push(c4); }
          let wants4 = !!(c4 && c3c4Tie);
          let wheelEligible = !c1c2Tie && c2c3Tie && (nLegs === 3 ? (wants4 && !c4c5Tie) : !wants4);

          let legs = wheelHorses.map(h => ({
            horse: h, p: exactaOrderProb(c1.PROGRAM, h.PROGRAM), pay: exactaPayoutEstimate(c1.PROGRAM, h.PROGRAM, baseU, baseU * wheelHorses.length)
          }));
          let allReal = legs.every(l => l.pay && l.pay.value > 0);
          if (allReal) {
            let cost = baseU * legs.length;
            let pWheel = legs.reduce((sum, l) => sum + l.p, 0);
            let expectedPayout = legs.reduce((sum, l) => sum + l.p * l.pay.value, 0) * baseU;
            let ev = expectedPayout - cost;
            let payValues = legs.map(l => l.pay.value);
            candidates.push({
              type: 'EXACTA_WHEEL', typeLabel: `Exacta Wheel (${legs.length + 1})`,
              horses: `#${c1.PROGRAM} / ${wheelHorses.map(h => '#' + h.PROGRAM).join(', ')}`,
              label: `$${cost.toFixed(0)} Exacta Wheel: #${c1.PROGRAM} over ${wheelHorses.map(h => '#' + h.PROGRAM).join(', ')}`,
              cost, hitProb: pWheel, ev, evPerDollar: ev / cost, eligible: wheelEligible,
              payoutLow: Math.min(...payValues) * baseU * 0.85, payoutHigh: Math.max(...payValues) * baseU * 1.15,
              isRealData: true,
              detail: `Live Probables, ${(pWheel * 100).toFixed(0)}% modeled hit rate (${legs.length} combos)`
            });
          }
        });
      }

      // --- Exacta Box (2 and 3 horse). Both sizes always generated; `eligible` = the original rule:
      // a box only makes sense when c1/c2 are near-tied, and extends to c3 only if c2/c3 are also tied. ---
      if (c1 && c2) {
        [2, 3].forEach(n => {
          if (n === 3 && !c3) return;
          let boxHorses = n === 3 ? [c1, c2, c3] : [c1, c2];
          let boxEligible = c1c2Tie && (n === 3 ? (c2c3Tie && !c3c4Tie) : !(c3 && c2c3Tie));

          let progs = boxHorses.map(h => h.PROGRAM);
          let combos = [];
          for (let i = 0; i < n; i++) {
            for (let j = 0; j < n; j++) {
              if (i !== j) combos.push([progs[i], progs[j]]);
            }
          }

          let cost = baseU * combos.length;
          let pBox = 0, expectedPayout = 0, allReal = true, payValues = [];
          combos.forEach(([a, b]) => {
            let p = exactaOrderProb(a, b);
            let pay = exactaPayoutEstimate(a, b, baseU, cost);
            if (!pay || !pay.isReal) allReal = false;
            pBox += p;
            expectedPayout += p * (pay ? pay.value : 0) * baseU;
            if (pay) payValues.push(pay.value);
          });

          if (allReal) {
            let ev = expectedPayout - cost;
            candidates.push({
              type: n === 2 ? 'EXACTA_BOX2' : 'EXACTA_BOX3', typeLabel: `Exacta Box (${n})`,
              horses: progs.map(p => `#${p}`).join(', '),
              label: `$${cost.toFixed(0)} Exacta Box (${n}): ${progs.map(p => '#' + p).join(', ')}`,
              cost, hitProb: pBox, ev, evPerDollar: ev / cost, eligible: boxEligible,
              payoutLow: (payValues.length ? Math.min(...payValues) : 0) * baseU * 0.85,
              payoutHigh: (payValues.length ? Math.max(...payValues) : 0) * baseU * 1.15,
              isRealData: true,
              detail: `Live Probables, ${(pBox * 100).toFixed(0)}% modeled hit rate (${combos.length} combos)`
            });
          }
        });
      }

      // --- Trifecta Box (3 and 4 horse). Both sizes always generated (when affordable under
      // MAX_TRIFECTA_STAKE and real data exists for every combo — a live trifecta pool total is NOT
      // required to display them); `eligible` = the original strict rule (which does require it): c1/c2/c3 mutually near-tied, 4th horse only if tied to c3, plus the hit-prob floor. ---
      if (c1 && c2 && c3) {
        let cost4Ok = (24 * TRIFECTA_MIN_BASE_UNIT) <= MAX_TRIFECTA_STAKE;
        let wants4 = !!(c4 && c3c4Tie && cost4Ok);
        [3, 4].forEach(n => {
          if (n === 3 && (6 * TRIFECTA_MIN_BASE_UNIT) > MAX_TRIFECTA_STAKE) return;
          if (n === 4 && (!c4 || !cost4Ok)) return;

          let boxProgs = (n === 4 ? [c1, c2, c3, c4] : [c1, c2, c3]).map(h => h.PROGRAM);
          let triCombos = [];
          for (let i = 0; i < n; i++) {
            for (let j = 0; j < n; j++) {
              for (let k = 0; k < n; k++) {
                if (i !== j && j !== k && i !== k) triCombos.push([boxProgs[i], boxProgs[j], boxProgs[k]]);
              }
            }
          }
          let triCost = triCombos.length * TRIFECTA_MIN_BASE_UNIT;
          let triPBox = 0, triExpectedPayout = 0, triAllReal = true;
          triCombos.forEach(([a, b, c]) => {
            let p = trifectaOrderProb(a, b, c);
            let pay = trifectaPayoutEstimate(a, b, c, TRIFECTA_MIN_BASE_UNIT, triCost);
            if (pay === null) triAllReal = false;
            triPBox += p;
            triExpectedPayout += p * (pay || 0) * TRIFECTA_MIN_BASE_UNIT;
          });

          if (triAllReal) {
            let triEligible = cachedTrifectaPoolTotal > 0 && c1c2Tie && c2c3Tie && (n === 4 ? (wants4 && !c4c5Tie) : !c3c4Tie) && triPBox >= MIN_TRIFECTA_HIT_PROB;
            let triEv = triExpectedPayout - triCost;
            candidates.push({
              type: 'TRIFECTA_BOX', typeLabel: `Trifecta Box (${n})`, horses: boxProgs.map(p => `#${p}`).join(', '),
              label: `$${triCost.toFixed(2)} Trifecta Box (${n}): ${boxProgs.map(p => '#' + p).join(', ')}`,
              cost: triCost, hitProb: triPBox, ev: triEv, evPerDollar: triEv / triCost, eligible: triEligible,
              payoutLow: triExpectedPayout * 0.6, payoutHigh: triExpectedPayout * 1.4,
              isRealData: true,
              detail: `Est. via real exacta anchor + modeled 3rd leg, ${(triPBox * 100).toFixed(0)}% modeled hit rate (${triCombos.length} combos)`
            });
          }
        });
      }

      // --- Custom wager: same helpers and payout models as the candidates above, but the user picks the
      // bet kind, the horses and the total amount. The combos are every ordered finish (distinct horses) the picks
      // allow, priced exactly like the engine's own straight / box / wheel candidates so the export reads the same.
      // Never part of Option A/B or the NOT RECOMMENDED rows. Returns { error } when the picks are incomplete or the
      // needed pool / Probables data isn't loaded yet. ---
      function priceCustomWager(spec) {
        let kindDef = CUSTOM_WAGER_KINDS.find(k => k.label === spec.kind);
        let amount = parseFloat(spec.amount);
        if (!kindDef) return { error: 'Choose a bet type.' };
        if (!(amount > 0)) return { error: 'Enter a wager amount.' };
        let n = kindDef.n;
        let live = list => (list || []).map(String).filter(p => winProbs[p]); // drop scratched / unknown horses
        let legs = [];
        if (kindDef.mode === 'box') {
          let box = live((spec.legs || [])[0]);
          if (box.length < n) return { error: 'Pick at least ' + n + ' horses for the box.' };
          for (let i = 0; i < n; i++) legs.push(box);
        } else {
          for (let i = 0; i < n; i++) legs.push(live((spec.legs || [])[i]));
          if (kindDef.mode === 'straight') legs = legs.map(l => l.slice(0, 1));
          if (legs.some(l => l.length === 0)) return { error: n === 1 ? 'Choose a horse.' : 'Choose at least one horse for each position (' + kindDef.slots.join(', ') + ').' };
        }
        let noData = { error: 'Pool or Probables data for this bet has not loaded yet.' };
        let tag = p => '#' + p;
        let uniq = a => a.filter((x, i) => a.indexOf(x) === i);
        let c = { cost: amount, isCustom: true, isRealData: true, legs };

        if (n === 1) {
          let p0 = legs[0][0];
          if (kindDef.label === 'Win') {
            let rate = getWinPoolPayout(p0, amount);
            if (rate === null) return noData;
            let payout = amount * rate;
            Object.assign(c, { type: 'WIN', typeLabel: 'Win', horses: tag(p0), hitProb: winProbs[p0].modelProb, payoutLow: payout, payoutHigh: payout, ev: winProbs[p0].modelProb * payout - amount });
          } else {
            let isPlace = kindDef.label === 'Place';
            let range = isPlace ? placeRateRange({ PROGRAM: p0 }, amount) : showRateRange({ PROGRAM: p0 }, amount);
            let pHit = isPlace ? placeProbabilityOf(p0) : showProbabilityOf(p0);
            if (range === null || !(pHit > 0)) return noData;
            Object.assign(c, { type: isPlace ? 'PLACE' : 'SHOW', typeLabel: kindDef.label, horses: tag(p0), hitProb: pHit, payoutLow: amount * range.min, payoutHigh: amount * range.max, ev: pHit * amount * range.avg - amount });
          }
        } else {
          let isTri = n === 3;
          let combos = [];
          if (n === 2) legs[0].forEach(a => legs[1].forEach(b => { if (a !== b) combos.push([a, b]); }));
          else legs[0].forEach(a => legs[1].forEach(b => legs[2].forEach(d => { if (a !== b && b !== d && a !== d) combos.push([a, b, d]); })));
          if (combos.length === 0) return { error: 'Pick at least two different horses.' };
          let pos = combos[0].map((_, i) => uniq(combos.map(x => x[i])));        // horses actually in play per position
          let typeLabel, type, horsesText;
          if (kindDef.mode === 'box') {
            let k = legs[0].length;
            typeLabel = (isTri ? 'Trifecta' : 'Exacta') + ' Box (' + k + ')';
            type = isTri ? 'TRIFECTA_BOX' : (k === 2 ? 'EXACTA_BOX2' : k === 3 ? 'EXACTA_BOX3' : 'EXACTA_BOX');
            horsesText = legs[0].map(tag).join(', ');
          } else if (kindDef.mode === 'straight') {
            typeLabel = isTri ? 'Trifecta (Straight)' : 'Exacta (Straight)';
            type = isTri ? 'TRIFECTA_CUSTOM' : 'EXACTA_STRAIGHT';
            horsesText = combos[0].map(tag).join('/');
          } else if (!isTri && pos[0].length === 1 && !pos[1].includes(pos[0][0])) {
            // key horse over the rest: named like the engine's own wheel
            typeLabel = 'Exacta Wheel (' + (pos[1].length + 1) + ')';
            type = 'EXACTA_WHEEL';
            horsesText = tag(pos[0][0]) + ' / ' + pos[1].map(tag).join(', ');
          } else {
            typeLabel = isTri ? 'Trifecta Wheel' : 'Exacta Wheel (Custom)';
            type = isTri ? 'TRIFECTA_CUSTOM' : 'EXACTA_CUSTOM';
            horsesText = pos.map(p => p.map(tag).join(', ')).join(' / ');
          }
          let unit = amount / combos.length;
          let pHit = 0, expected = 0, pays = [];
          for (let combo of combos) {
            let pay = isTri ? trifectaPayoutEstimate(combo[0], combo[1], combo[2], unit, amount) : getExactaPayout(combo[0], combo[1], unit, amount);
            if (pay === null || !(pay > 0)) return noData;
            let p = isTri ? trifectaOrderProb(combo[0], combo[1], combo[2]) : exactaOrderProb(combo[0], combo[1]);
            pHit += p; expected += p * pay * unit; pays.push(pay);
          }
          let low, high;
          if (isTri) { low = expected * 0.6; high = expected * 1.4; }
          else { low = Math.min(...pays) * unit * 0.85; high = Math.max(...pays) * unit * 1.15; }
          Object.assign(c, { type, typeLabel, horses: horsesText, hitProb: pHit, payoutLow: low, payoutHigh: high, ev: expected - amount, unit, combos: combos.length });
        }
        c.evPerDollar = c.ev / amount;
        return c;
      }
      customWagerResult = priceCustomWager(customWager);
    }

    // Only candidates that pass the clustering rules compete for Option A/B; the rest feed the expanded table.
    let eligibleCands = candidates.filter(c => c.eligible);
    let positiveEv = eligibleCands.filter(c => c.evPerDollar >= MIN_EV_PER_DOLLAR && c.cost > 0); // (name kept; = "acceptable EV")
    // Payout ratio uses the conservative (low) end of the estimated range, so a candidate only
    // clears the bar if it's still worth it even under the more pessimistic payout estimate.
    let worthwhile = positiveEv.filter(c => (c.payoutLow / c.cost) >= MIN_PAYOUT_RATIO);
    let scenario = "";
    let optionRows = [];

    if (positiveEv.length === 0) {
      // "Too Close" = the engine generated nothing at all (near-ties / missing data blocked every bet).
      // "Low Payout" = bets existed but none had a positive edge at the posted payouts.
      let noBetReason = eligibleCands.length === 0 ? 'No Bet - Too Close' : 'No Bet - Low Payout';
      scenario = liveRunners.length < 2 ? "Insufficient Live Runners"
        : eligibleCands.length === 0 ? "No Wager Identified — field too closely matched to find a real edge"
        : `No Wager Above the ${(MIN_EV_PER_DOLLAR * 100).toFixed(0)}% EV Floor — posted payouts too low for the modeled hit rates`;
      optionRows = [
        { label: 'A', isPass: true, passReason: noBetReason },
        { label: 'B', isPass: true, passReason: noBetReason }
      ];
    } else if (worthwhile.length === 0) {
      scenario = `EV Acceptable Only on Thin-Margin Favorites — payout under ${MIN_PAYOUT_RATIO}x the stake`;
      optionRows = [
        { label: 'A', isPass: true, passReason: 'No Bet - Low Payout' },
        { label: 'B', isPass: true, passReason: 'No Bet - Low Payout' }
      ];
    } else {
      let byEv = worthwhile.slice().sort((a, b) => b.evPerDollar - a.evPerDollar);
      let best = byEv[0];
      let byHitRate = worthwhile.slice().sort((a, b) => b.hitProb - a.hitProb).find(c => c.type !== best.type) || byEv[1];
      best.optionLabel = 'A'; // tags the chosen candidates for the Research Export's bet table
      if (byHitRate) byHitRate.optionLabel = 'B';

      scenario = `Value Pick: ${best.type.replace('_', ' ')} (EV/$ = ${best.evPerDollar >= 0 ? '+' : ''}${(best.evPerDollar * 100).toFixed(0)}%)` +
        (byHitRate ? ` | Safety Pick: ${byHitRate.type.replace('_', ' ')} (${(byHitRate.hitProb * 100).toFixed(0)}% modeled hit rate)` : '');

      optionRows = [
        { label: 'A', isPass: false, stake: best.cost, typeLabel: best.typeLabel, horses: best.horses, payoutLow: best.payoutLow, payoutHigh: best.payoutHigh, hitProb: best.hitProb, isRealData: best.isRealData },
        byHitRate
          ? { label: 'B', isPass: false, stake: byHitRate.cost, typeLabel: byHitRate.typeLabel, horses: byHitRate.horses, payoutLow: byHitRate.payoutLow, payoutHigh: byHitRate.payoutHigh, hitProb: byHitRate.hitProb, isRealData: byHitRate.isRealData }
          : { label: 'B', isPass: true, passReason: 'No Bet - Low Payout' }
      ];
    }

    // Expanded-table rows: best candidate (by EV/$, same ranking as Option A) for every bet type/
    // configuration the engine generated, minus the types already used by Option A/B.
    let usedTypeLabels = new Set(optionRows.filter(o => !o.isPass).map(o => o.typeLabel));
    let bestPerType = {};
    candidates.forEach(c => {
      if (!(c.cost > 0)) return;
      let cur = bestPerType[c.typeLabel];
      if (!cur || c.evPerDollar > cur.evPerDollar) bestPerType[c.typeLabel] = c;
    });
    let extraRows = Object.values(bestPerType)
      .filter(c => !usedTypeLabels.has(c.typeLabel))
      .sort((a, b) => b.hitProb - a.hitProb)
      .map(c => ({ label: 'NOT RECOMMENDED', isExtra: true, isPass: false, stake: c.cost, typeLabel: c.typeLabel, horses: c.horses, payoutLow: c.payoutLow, payoutHigh: c.payoutHigh, hitProb: c.hitProb, isRealData: c.isRealData }));

    return {
      ruleAppliedMsg,
      leaderboard: runners,
      gaps: { gap12, gap23, gap34, gap45, gap13, gap14 },
      wagerRecs: { scenario, optionRows, extraRows },
      wetnessTier,
      trackConditionRaw: cachedTrackCondition,
      weightKey: activeWeightKey,
      allCandidates: candidates,
      customWager: customWagerResult
    };
  }

  // ===== RESULTS (official race) =====
  // Shape confirmed from a real Results payload: raceDetails.{raceNumber, dataStatus, finishOrder[],
  // alsoRans[], wpsMatrix[], exoticPools[]}. WPS payouts are per $2; exotic `value` is per `base`
  // (EX base 1.0, TR base 0.5, QD base 0.1), and `result` is the winning sequence, e.g. "5/1/2".
  function findResultsPayload(obj, depth = 0) {
    if (!obj || typeof obj !== 'object' || depth > 5) return null;
    if (obj.raceDetails && Array.isArray(obj.raceDetails.finishOrder)) return obj;
    if (Array.isArray(obj)) {
      for (let item of obj) {
        let found = findResultsPayload(item, depth + 1);
        if (found) return found;
      }
    } else {
      for (let key in obj) {
        if (typeof obj[key] === 'object' && obj[key] !== null) {
          let found = findResultsPayload(obj[key], depth + 1);
          if (found) return found;
        }
      }
    }
    return null;
  }

  function parseResultsPayload(payload) {
    let rd = payload.raceDetails || {};
    let finish = {};   // program -> finish position
    (rd.finishOrder || []).concat(rd.alsoRans || []).forEach(f => {
      let prog = String(f.programNumber);
      let pos = parseInt(f.finishPosition, 10);
      if (prog && !isNaN(pos) && finish[prog] === undefined) finish[prog] = pos;
    });
    let wps = {};      // program -> { win, place, show } payouts per $2 (null where the horse didn't pay that pool)
    (rd.wpsMatrix || []).forEach(w => {
      wps[String(w.programNumber)] = {
        win: typeof w.winPayout === 'number' ? w.winPayout : null,
        place: typeof w.placePayout === 'number' ? w.placePayout : null,
        show: typeof w.showPayout === 'number' ? w.showPayout : null
      };
    });
    let exotics = {};  // poolCode -> { value, base, result: [programs in finishing order] }
    (rd.exoticPools || []).forEach(x => {
      exotics[String(x.poolCode)] = {
        value: parseFloat(x.value), base: parseFloat(x.base) || 1,
        result: String(x.result || '').split('/').map(s => s.trim()).filter(Boolean)
      };
    });
    // The finishOrder / alsoRans lists are not always filled in (confirmed: a payload had both as empty
    // arrays while still reporting COMPLETE). The finishing order is also carried by finishOrderString
    // ("7/4/1/6"), by the exotic pool results (the longest one, e.g. the superfecta, "7/4/1/6"), and by
    // wpsMatrix, which lists the first three finishers in seq order. Use those, in that order, only when
    // the lists above yielded nothing.
    let finishSource = 'finishOrder';
    if (Object.keys(finish).length === 0) {
      let order = String(rd.finishOrderString || '').split('/').map(x => x.trim()).filter(Boolean);
      finishSource = 'finishOrderString';
      if (order.length === 0) {
        Object.values(exotics).forEach(x => { if (x.result.length > order.length) order = x.result.slice(); });
        finishSource = 'longest exotic result';
      }
      if (order.length === 0) {
        order = (rd.wpsMatrix || []).slice().sort((a, b) => (a.seq || 0) - (b.seq || 0)).map(w => String(w.programNumber));
        finishSource = 'wpsMatrix';
      }
      order.forEach((prog, i) => { if (prog && finish[prog] === undefined) finish[prog] = i + 1; });
    }

    // Shape summary shown in the panel when results load but yield no finishers, so the real payload
    // layout can be seen without opening DevTools: entry counts plus the field names of the first entry.
    let fo0 = (rd.finishOrder && rd.finishOrder[0]) || (rd.alsoRans && rd.alsoRans[0]) || {};
    let diag = `finishOrder:${(rd.finishOrder || []).length} alsoRans:${(rd.alsoRans || []).length} wps:${(rd.wpsMatrix || []).length} exotics:${(rd.exoticPools || []).length} firstEntryFields:[${Object.keys(fo0).join(',')}] finishFrom:${finishSource}`;
    return {
      raceNumber: parseInt(rd.raceNumber, 10),
      track: String((payload.track && (payload.track.displayName || payload.track.brisCode)) || ''),
      status: String(rd.dataStatus || ''),
      finish, wps, exotics, diag
    };
  }

  // The parsed results for the race currently on screen, only if they are complete (race number must
  // match; track name is compared loosely since display names vary).
  function getOfficialResults() {
    let norm = s => String(s || '').toLowerCase().replace(/[^a-z0-9]/g, '');
    let curRace = parseInt(getActiveRaceNum(), 10);
    let curTrack = norm(getActiveTrackName());
    // A payload can sit at PENDING (confirmed: the Results tab showed no Finish Order and the payload reported PENDING
    // while the race itself was OFFICIAL and the payouts were posted). Treat PENDING as usable only when a win payout is
    // actually present, which only exists once the race has been run and the tote has paid.
    let hasWinPayout = res => Object.values(res.wps).some(w => typeof w.win === 'number');
    return cachedResultsList.find(res => {
      let status = res.status.toUpperCase();
      if (!(status === 'COMPLETE' || (status === 'PENDING' && hasWinPayout(res))) || Object.keys(res.finish).length === 0) return false;
      if (res.raceNumber !== curRace) return false;
      let a = norm(res.track);
      return !(a && curTrack && !(a.includes(curTrack) || curTrack.includes(a)));
    }) || null;
  }

  // Plain-text explanation of why loaded results didn't match the race on screen (shown in the

  function describeResultsMismatch() {
    let loaded = cachedResultsList.map(r => `R${r.raceNumber} ${r.track || '?'} [${r.status || 'no status'}, ${Object.keys(r.finish).length} finishers; ${r.diag}]`).join('; ');

  // Did this recommended bet win, and what did it actually return (stake included)? Uses the real
  // official payouts. Returns { hit, payout } where payout is null if it hit but the needed pool
  // payout isn't in the results (left blank rather than guessed).
  function getBetOutcome(c, res) {
    let progs = (String(c.horses).match(/#[A-Za-z0-9]+/g) || []).map(s => s.slice(1));
    let pos = p => res.finish[String(p)];
    let byPos = n => Object.keys(res.finish).filter(p => res.finish[p] === n)[0];
    let ex = res.exotics.EX, tr = res.exotics.TR;
    let exUnit = c.unit || cachedExactaBaseUnit || 1;   // per-combo cost the model used for exacta bets
    // Winning sequences come from the pools' own result strings when present (so ties/dead heats
    // follow what the tote paid), otherwise from the finish order.
    let seq2 = (ex && ex.result.length >= 2) ? ex.result : [byPos(1), byPos(2)];
    let seq3 = (tr && tr.result.length >= 3) ? tr.result : [byPos(1), byPos(2), byPos(3)];
    let w = res.wps[progs[0]] || {};
    let perUnit = pool => (pool && isFinite(pool.value)) ? pool.value / pool.base : null;
    let hit = false, payout = null;

    switch (c.type) {
      case 'WIN':   hit = pos(progs[0]) === 1; payout = hit && w.win !== null && w.win !== undefined ? w.win * c.cost / 2 : null; break;
      case 'PLACE': hit = pos(progs[0]) >= 1 && pos(progs[0]) <= 2; payout = hit && w.place != null ? w.place * c.cost / 2 : null; break;
      case 'SHOW':  hit = pos(progs[0]) >= 1 && pos(progs[0]) <= 3; payout = hit && w.show != null ? w.show * c.cost / 2 : null; break;
      case 'EXACTA_STRAIGHT':
        hit = seq2[0] === progs[0] && seq2[1] === progs[1];
        payout = hit && perUnit(ex) !== null ? perUnit(ex) * c.cost : null; break;
      case 'EXACTA_WHEEL':
        hit = seq2[0] === progs[0] && progs.slice(1).includes(seq2[1]);
        payout = hit && perUnit(ex) !== null ? perUnit(ex) * exUnit : null; break;
      case 'EXACTA_BOX2':
      case 'EXACTA_BOX3':
      case 'EXACTA_BOX':
        hit = progs.includes(seq2[0]) && progs.includes(seq2[1]);
        payout = hit && perUnit(ex) !== null ? perUnit(ex) * exUnit : null; break;
      case 'EXACTA_CUSTOM': // user-built part wheel: first and second must come from the matching position lists
        hit = c.legs[0].includes(seq2[0]) && c.legs[1].includes(seq2[1]);
        payout = hit && perUnit(ex) !== null ? perUnit(ex) * exUnit : null; break;
      case 'TRIFECTA_CUSTOM':
        hit = c.legs.every((leg, i) => leg.includes(seq3[i]));
        payout = hit && perUnit(tr) !== null ? perUnit(tr) * (c.unit || TRIFECTA_MIN_BASE_UNIT) : null; break;
      case 'TRIFECTA_BOX':
        hit = seq3.every(p => progs.includes(p));
        payout = hit && perUnit(tr) !== null ? perUnit(tr) * (c.unit || TRIFECTA_MIN_BASE_UNIT) : null; break;
      default: return null;
    }
    return { hit, payout: hit ? payout : 0 };
  }

  // The No Bet row the user ticked (still a pass row right now), or null.
  function getUserSelectedNoBetRow(modelRes) {
    if (!userSelectedWagerKey || userSelectedWagerKey.indexOf(NO_BET_KEY_PREFIX) !== 0) return null;
    let opts = (modelRes.wagerRecs && modelRes.wagerRecs.optionRows) || [];
    return opts.find(o => o.isPass && NO_BET_KEY_PREFIX + o.label === userSelectedWagerKey) || null;
  }
  // Selected wager text (bets export + panel). With nothing ticked it stays blank, except on a race
  // where the model recommended no bet at all, which records that reason.
  function getUserSelectedWagerText(modelRes) {
    let sel = getUserSelectedCandidate(modelRes);
    if (sel) return formatWagerLabel(sel);
    let noBetRow = getUserSelectedNoBetRow(modelRes);
    if (noBetRow) return noBetRow.passReason;
    let opts = (modelRes.wagerRecs && modelRes.wagerRecs.optionRows) || [];
    return (opts.length > 0 && opts.every(o => o.isPass)) ? opts[0].passReason : '';
  }

  function getResearchRaceContext() {
    let rawPostTime = getActivePostTime();
    let { dateStr, timeStr } = parseExportDateTime(rawPostTime);
    let dp = dateStr.split('/');
    let isoDate = `${dp[2]}-${dp[0].padStart(2, '0')}-${dp[1].padStart(2, '0')}`;
    let trackName = getActiveTrackName().replace(/"/g, '');
    let raceNum = getActiveRaceNum();
    let postMs = new Date(rawPostTime).getTime();
    let minToPost = (rawPostTime && rawPostTime !== 'N/A' && !isNaN(postMs)) ? Math.max(0, (postMs - Date.now()) / 60000).toFixed(1) : ''; // bottoms out at 0 (race already off / official)
    return { raceId: `${isoDate}_${trackName}_${raceNum}`, dateStr, timeStr, trackName, raceNum, minToPost, snapshotTime: new Date().toTimeString().slice(0, 8) };
  }

  function getResearchHorseRows(modelRes) {
    let ctx = getResearchRaceContext();
    let activeHorses = modelRes.leaderboard.filter(h => !h.IS_SCRATCHED && !cachedScratchedMap[h.PROGRAM]);
    let weekRaces = cachedStatsObj ? getVal(cachedStatsObj, ['weekCountRaces', 'weekRaces'], '') : '';
    let meetRaces = cachedStatsObj ? getVal(cachedStatsObj, ['meetCountRaces', 'meetRaces', 'meetCount'], '') : '';
    // Race-level attributes live on the same object as the meet/week stats (the payload's `race`
    // object). distance + distanceUnit are exported raw (unit seen so far: F with distance 850 = 8.5
    // furlongs) rather than converted, since other breeds/units haven't been sampled.
    let rs = cachedStatsObj;
    let raceCells = [
      ctx.raceId, ctx.dateStr, ctx.timeStr, ctx.snapshotTime, ctx.minToPost, ctx.trackName, ctx.raceNum,
      cachedBreed || 'Thoroughbred', MODEL_VERSION, modelRes.weightKey || '', activeHorses.length,
      cachedTrackCondition || '', modelRes.wetnessTier !== undefined ? modelRes.wetnessTier : '',
      rs ? getVal(rs, 'distance', '') : '', rs ? getStr(rs, 'distanceUnit', '') : '', rs ? getStr(rs, 'distanceFormatted', '') : '',
      rs ? getStr(rs, 'surface', '') : '', rs ? getStr(rs, 'raceType', '') : '', rs ? getStr(rs, 'raceClassification', '') : '',
      rs ? getVal(rs, 'purse', '') : '', rs ? getVal(rs, 'claimingPrice', '') : '', rs ? getVal(rs, 'speedPar', '') : '',
      weekRaces, meetRaces
    ];
    // The two after-the-fact inputs are race-level, so every horse row of the race carries the same
    // values: WINNER (program # of the winner) and TOP3 (comma-separated program #s of the first three
    // finishers, in finishing order). Filled in automatically once the race is official and its results
    // have loaded; blank otherwise, to be typed in by hand.
    let res = getOfficialResults();
    let winnerProg = '', top3Progs = '';
    if (res) {
      let finishers = Object.keys(res.finish).sort((a, b) => res.finish[a] - res.finish[b]);
      winnerProg = finishers[0] || '';
      top3Progs = finishers.slice(0, 3).join(',');
    }
    return activeHorses.map(h => {
      let liveDec = parseOddsToDecimal(h.LIVE_ODDS);
      // Column order matches the workbook: RACE_ID, WINNER, TOP3, then the rest.
      return tsvRow([raceCells[0], winnerProg, top3Progs].concat(raceCells.slice(1), [
        h.PROGRAM, h.HORSE_NAME.replace(/"/g, ''), h.POST, h.RUN_STYLE, h.RUN_STYLE_PTS,
        numOrBlank(h.ODDS_NUMERIC, 2), (!isNaN(liveDec) && liveDec > 0) ? 'LIVE' : 'ML',
        numOrBlank(h.C_SPEED, 3), numOrBlank(h.C_POWER, 3), numOrBlank(h.C_CLASS, 3), numOrBlank(h.C_DRIVER, 4),
        numOrBlank(h.C_TRAINER, 4), numOrBlank(h.C_EARLY, 3), numOrBlank(h.C_FINISH, 3), numOrBlank(h.C_RECENCY, 1), numOrBlank(h.C_MARKET, 3),
        numOrBlank(h.POST_BONUS, 2), numOrBlank(h.STYLE_BONUS, 2), numOrBlank(h.WET_BONUS, 2),
        numOrBlank(h.FINAL_SCORE, 2), h.RANK, numOrBlank(h.Z_SCORE, 3), numOrBlank(h.WIN_PROB, 4), h.IMPUTED_COUNT || 0
      ]));
    }).join('\n');
  }

  function getResearchBetRows(modelRes) {
    let ctx = getResearchRaceContext();
    let derived = getDerivedTakeout();
    // Column order matches the workbook: RACE_ID, HIT, ACTUAL_PAYOUT, then the rest. HIT and
    // ACTUAL_PAYOUT are filled in automatically once the race is official; blank otherwise.
    let res = getOfficialResults();
    let raceCells = [ctx.dateStr, ctx.trackName, ctx.raceNum, cachedBreed || 'Thoroughbred', MODEL_VERSION];
    let poolCells = [
      cachedWinPoolTotal || 0, cachedPlacePoolTotal || 0, cachedShowPoolTotal || 0, cachedExactaPoolTotal || 0, cachedTrifectaPoolTotal || 0,
      getEffectiveTakeout().toFixed(4), derived.rate !== null ? derived.rate.toFixed(4) : ''
    ];
    let picked = (modelRes.allCandidates || []).filter(c => c.optionLabel === 'A' || c.optionLabel === 'B')
      .sort((a, b) => a.optionLabel.localeCompare(b.optionLabel));
    // A ticked wager that isn't Option A/B (a NOT RECOMMENDED row) still needs a row here, so its
    // hit/payout can be looked up for the results sheet. It is tagged OPTION = USER.
    let userPick = getUserSelectedCandidate(modelRes);
    if (userPick && !picked.includes(userPick)) picked.push(Object.assign({}, userPick, { optionLabel: 'USER', isUserPick: true }));
    if (picked.length === 0) {
      let reasonRow = ((modelRes.wagerRecs && modelRes.wagerRecs.optionRows) || []).find(o => o.isPass);
      // No bet was recommended, so there is nothing to score: HIT and ACTUAL_PAYOUT stay blank.
      return tsvRow([ctx.raceId, '', ''].concat(raceCells, ['', reasonRow ? reasonRow.passReason : 'No Bet', '', '', '', '', '', ''], poolCells, [getUserSelectedWagerText(modelRes)])); // no bet recommended: the no-bet reason
    }
    let noBetPick = getUserSelectedNoBetRow(modelRes);
    return picked.map(c => {
      let outcome = res ? getBetOutcome(c, res) : null;
      let hitCell = outcome ? (outcome.hit ? 'Y' : 'N') : '';
      let payCell = (outcome && outcome.payout !== null) ? outcome.payout.toFixed(2) : '';
      return tsvRow([ctx.raceId, hitCell, payCell].concat(raceCells, [
        c.optionLabel, c.typeLabel, c.horses, numOrBlank(c.cost, 2), numOrBlank(c.hitProb, 4),
        numOrBlank(c.evPerDollar, 4), numOrBlank(c.payoutLow, 2), numOrBlank(c.payoutHigh, 2)
      ], poolCells, [(c === userPick || c.isUserPick) ? formatWagerLabel(c) : '']));  // USER_SELECTED_WAGER: filled only on the ticked bet's own row (an Option A/B pick keeps its A/B row)
    }).concat(noBetPick ? [tsvRow([ctx.raceId, '', ''].concat(raceCells, ['USER', noBetPick.passReason, '', '', '', '', '', ''], poolCells, [noBetPick.passReason]))] : []).join('\n'); // the user chose No Bet although another option was a real bet
  }

  function updateOverlay(force) {
    applyOverlayThemeCSS();
    renderToggleButton();
    let existing = document.getElementById('ts-model-overlay');
    if (!existing) {
      existing = document.createElement('div');
      existing.id = 'ts-model-overlay';
      existing.style.cssText = 'position:fixed;top:40px;right:20px;z-index:999999;padding:16px;border-radius:10px;box-shadow:0 10px 25px rgba(0,0,0,0.5);width:min(' + currentWidth + 'px, 94vw);min-width:300px;max-height:' + currentHeight + 'px;overflow-y:auto;border:2px solid var(--ts-purple);transform-origin:top right;';
      document.body.appendChild(existing);
    }

    existing.style.display = isOverlayVisible ? 'block' : 'none';
    existing.classList.toggle('ts-dark', isDarkMode);

    // A re-render replaces every element, which would close an open dropdown or drop the cursor from the amount box.
    // So while a custom-wager control has focus and was touched in the last 10s, hold the refresh; it runs when focus leaves.
    let focused = document.activeElement;
    if (force !== true && focused && focused.classList && focused.classList.contains('ts-custom-ctl') && existing.contains(focused) && (Date.now() - customWagerLastTouch) < 10000) {
      customRenderDeferred = true;
      return;
    }
    customRenderDeferred = false;

    let trackDisplay = getActiveTrackName().toUpperCase().replace(/"/g, '');
    let raceDisplay = String(getActiveRaceNum()).replace(/[^0-9]/g, '');
    let activeStartersCount = Object.values(cachedHorsesMap).filter(h => !h.IS_SCRATCHED && !cachedScratchedMap[h.PROGRAM]).length;
    let scratchedHorsesCount = Object.values(cachedHorsesMap).filter(h => h.IS_SCRATCHED || cachedScratchedMap[h.PROGRAM]).length;
    let currentDate = new Date().toISOString().split('T')[0];
    let condTag = cachedTrackCondition ? ` (${cachedTrackCondition.toUpperCase()})` : '';

    let modelRes = calculateModelOutput();

    // Low-data warning: a normal race's top score lands well above 60; a top score below that means whole
    // scoring components are empty across the field (e.g. a field of first-time starters), so the ranking
    // is close to random. Requires a nonzero score so it doesn't fire while data is still loading.
    let topActiveScore = Math.max(0, ...modelRes.leaderboard
      .filter(h => !h.IS_SCRATCHED && !cachedScratchedMap[h.PROGRAM])
      .map(h => h.FINAL_SCORE || 0));

        </div>` : '';

      if (tracked && tracked.rankChangedAt && (now - tracked.rankChangedAt) < CHANGE_INDICATOR_DURATION_MS) {
        rankArrow = tracked.rankDir === 'up' ? '▲' : '▼';

  // True when a runner/stats payload names a race number that differs from the race in the URL, so another
  // race's horses (same program numbers) can't merge into this race's data. Only fires when BOTH are known.
  function isOtherRacePayload(urlData, payloadRace) {
    if (!urlData || !payloadRace) return false;
    let p = String(payloadRace).replace(/[^0-9]/g, '');
    return p !== '' && parseInt(p, 10) !== urlData.raceNum;
  }

  function processResponse(url, text, seq) {
    if (seq !== undefined) {
      let pattern = getUrlPattern(url);
      if (lastSeqByUrlPattern[pattern] !== undefined && seq < lastSeqByUrlPattern[pattern]) return; // stale, superseded
      lastSeqByUrlPattern[pattern] = seq;
    }
    if (!text || (text.trim().charAt(0) !== '{' && text.trim().charAt(0) !== '[')) return;
    try {
      let data = JSON.parse(text);

      let payloadCond = getStr(data, ['trackCondition', 'condition', 'surfaceCondition', 'track_condition'], '');
      if (payloadCond) cachedTrackCondition = payloadCond;

      let payloadPostTime = getStr(data, [
        'postTime', 'post_time', 'startTime', 'racePostTime', 
        'postTimeFormatted', 'post_time_formatted', 'scheduledStart', 
        'postTimeEpoch', 'postTimeLocal', 'postTimeText'
      ], '');
      if (payloadPostTime) cachedPostTime = payloadPostTime;

      // Runs BEFORE this payload merges, so a reset never wipes the fresh data that triggered it.
      let urlData = parseTwinSpiresUrl();
      if (urlData) checkRaceChange(urlData.trackName, urlData.raceNum);

      // A Results payload also contains arrays of objects with a programNumber (winPools, wpsMatrix,
      // finishOrder...), which findRunnersArray would mistake for runner rows — so capture it first
      // and skip runner parsing for it.
      let resultsPayload = findResultsPayload(data);
      if (resultsPayload) {
        let parsedResults = parseResultsPayload(resultsPayload);
        // keep one entry per race (replace an earlier copy of the same race, e.g. a status refresh)
        cachedResultsList = cachedResultsList.filter(r => !(r.raceNumber === parsedResults.raceNumber && r.track === parsedResults.track)).concat(parsedResults);
        updateOverlay();
      }

      let runners = resultsPayload ? null : findRunnersArray(data);
      if (runners && runners.length > 0 && !isOtherRacePayload(urlData, runners[0].raceNumber || data.raceNumber)) {
        let sample = runners[0];
        let payloadTrack = getActiveTrackName(sample) || getActiveTrackName(data);
        let payloadRace = sample.raceNumber || data.raceNumber;

        if (!urlData) checkRaceChange(payloadTrack, payloadRace); // payload identity only when the URL can't be parsed
        if (payloadTrack && payloadTrack !== "Unknown Track") cachedTrackName = payloadTrack;
        if (payloadRace) cachedRaceNum = payloadRace;

        runners.forEach(r => parseAndMergeRunnerRow(r));
        updateOverlay();
      }

      let statsObj = findStatsObject(data);
      if (statsObj && !isOtherRacePayload(urlData, statsObj.raceNumber || data.raceNumber)) {
        cachedStatsObj = statsObj;
        let payloadTrack = getActiveTrackName(statsObj) || getActiveTrackName(data);
        let payloadRace = statsObj.raceNumber || data.raceNumber;

        if (!urlData) checkRaceChange(payloadTrack, payloadRace); // payload identity only when the URL can't be parsed
        if (payloadTrack && payloadTrack !== "Unknown Track") cachedTrackName = payloadTrack;
        if (payloadRace) cachedRaceNum = payloadRace;

        let statRows = parseStatsRows(statsObj);
        let statsHeader = Object.keys(statRows[0]).join(',');
        cachedStatsCsv = `${statsHeader}\n${statRows.map(r => Object.values(r).join(',')).join('\n')}`;
        updateOverlay();
      }

      let exoticsPayload = findExoticsPayload(data);
      if (exoticsPayload) {
        let parsed = parseExoticsGrid(exoticsPayload);
        cachedExactaGrid = parsed.grid;
        cachedExactaPoolTotal = parsed.poolTotal;
        cachedExactaBaseUnit = parsed.baseUnit;
        cachedTrifectaPoolTotal = parsed.trifectaPoolTotal;
        cachedTrifectaBaseUnit = parsed.trifectaBaseUnit;
        loadedTabs['Probables'] = true;
        updateOverlay();
      }

      let poolsPayload = findPoolsPayload(data);
      if (poolsPayload) {
        let parsedPools = parsePoolsPayload(poolsPayload);
        cachedPoolsMap = parsedPools.poolsMap;
        cachedWinPoolTotal = parsedPools.winTotal;
        cachedPlacePoolTotal = parsedPools.placeTotal;
        cachedShowPoolTotal = parsedPools.showTotal;
        cachedShowPoolAvailable = parsedPools.showAvailable;
        cachedPoolsWasPercentMode = parsedPools.detectedPercentMode;
        loadedTabs['Pools'] = true;
        updateOverlay();
      }
    } catch(e) {}
  }

W_Power + Class x W_Class\n     + Driver x W_Driver + Trainer x W_Trainer + Early x W_Early + Finish x W_Finish\n     + Recency x W_Recency + Market x W_Market\n</code></pre>\n<h3>The weights</h3>\n<p>Different breeds are won in different ways, so each has its own weight set:</p>\n<div class=\"tw\"><table>\n<thead>\n<tr>\n<th>Weight</th>\n<th>Thoroughbred</th>\n<th>Harness</th>\n<th>Quarter Horse</th>\n<th><strong>Special</strong> (Thoroughbred, selected tracks)</th>\n</tr>\n</thead>\n<tbody>\n<tr>\n<td>Speed</td>\n<td>0.28</td>\n<td>0.20</td>\n<td>0.28</td>\n<td>0.25</td>\n</tr>\n<tr>\n<td>Power</td>\n<td>0.09</td>\n<td>0.07</td>\n<td>0.10</td>\n<td>0.08</td>\n</tr>\n<tr>\n<td>Class</td>\n<td>0.16</td>\n<td>0.13</td>\n<td>0.09</td>\n<td>0.15</td>\n</tr>\n<tr>\n<td>Driver</td>\n<td>0.04</td>\n<td>0.16</td>\n<td>0.05</td>\n<td>0.04</td>\n</tr>\n<tr>\n<td>Trainer</td>\n<td>0.07</td>\n<td>0.05</td>\n<td>0.06</td>\n<td>0.08</td>\n</tr>\n<tr>\n<td>Early pace</td>\n<td>0.15</td>\n<td>0.18</td>\n<td>0.34</td>\n<td>0.14</td>\n</tr>\n<tr>\n<td>Finish (late pace)</td>\n<td>0.10</td>\n<td>0.08</td>\n<td>0.00</td>\n<td>0.11</td>\n</tr>\n<tr>\n<td>Recency</td>\n<td>0.03</td>\n<td>0.04</td>\n<td>0.02</td>\n<td>0.03</td>\n</tr>\n<tr>\n<td>Market</td>\n<td>0.03</td>\n<td>0.03</td>\n<td>0.03</td>\n<td>0.09</td>\n</tr>\n</tbody>\n</table></div>\n<p>Reading the table: harness leans on the driver and early speed; quarter horses are mostly early speed and raw speed\n(late pace is ignored).</p>\n<p>A separate Distance component was removed: it was almost the same number as Speed (a 0.97 correlation, since Speed already\nblends in the distance/surface speed figure), so it was double counting. Its old weight was added to Speed as a stand-in until\nthe weights are fitted.</p>\n<p><strong>Special weights</strong> are used only for <strong>Thoroughbred races at Churchill Downs and Horseshoe Indianapolis</strong>. They\ndiffer mainly by trusting the market more (Market weight 0.09 vs 0.03). This list is <strong>hardcoded on purpose</strong>.\nIt is never decided from live data such as pool size, because pool totals can nearly double in the final minute\nbefore post, which would silently flip the weight set after someone had already decided on a bet. A track is only\nadded after it has been tested independently.</p>\n<h3>Why the weights do not sum to exactly 1.0</h3>\n<p>The sets sum to about 0.94 to 0.97. This does not matter, because the next stage converts scores to\n<strong>relative standing within the field</strong> (z-scores). A uniform scale change has no effect on the outcome. What matters is the\n<em>proportion</em> between weights.</p>\n<p>The weights are <strong>not fitted to results yet</strong>; they are judgment-based starting values. Components also spread horses\napart by different amounts (class figures are bunched tightly, market odds are not), so a weight is not the same thing as\nthat component's real influence on the ranking. Driver and trainer win rates are converted to percentage points before\nweighting so they sit on a comparable scale to the other figures; even so, their influence is modest.</p>\n<hr />\n<h2>6. Stage 3c: Bonuses (the \"on top of the score\" layer)</h2>\n<p>Three small, capped adjustments are added to the base score to get the <strong>final score</strong>. They use <strong>track-bias data</strong>\nfrom the Meet and Week stats on TwinSpires.</p>\n<h3>6.1 Which stats to trust: Week vs Meet</h3>\n<p>Bias data comes in two flavors, this week and the whole meet. A short week is noisy, so they are blended by sample\nsize:</p>\n<div class=\"tw\"><table>\n<thead>\n<tr>\n<th>Races this week</th>\n<th>Blend used</th>\n</tr>\n</thead>\n<tbody>\n<tr>\n<td>15 or more</td>\n<td>65% Week, 35% Meet</td>\n</tr>\n<tr>\n<td>7 to 14</td>\n<td>20% Week, 80% Meet</td>\n</tr>\n<tr>\n<td>Fewer than 7</td>\n<td>100% Meet</td>\n</tr>\n</tbody>\n</table></div>\n<h3>6.2 Is there enough meet data at all?</h3>\n<p>Bonuses are skipped entirely if the meet has <strong>fewer than 15 races</strong> of data. Early-pace weight changes also scale\nwith meet size (see 6.3).</p>\n<h3>6.3 Speed-bias adjustment (changes a weight, not a bonus)</h3>\n<p>If the track is playing strongly to speed (blended speed bias above 0.55), the <strong>Early weight is increased</strong> for\nthis race: <code>W_Early += (bias - 0.55) x 0.5</code>, scaled down for small meet samples (0% under 6 races, 10% from 6 to 15,\n100% above 15). A speed-favoring track therefore makes early pace count for more in everyone's score.</p>\n<h3>6.4 Post-position bonus</h3>\n<p>Posts are grouped: <strong>Rail</strong> (1), <strong>1-3</strong>, <strong>4-7</strong>, <strong>8+</strong>. The track's \"impact value\" for the horse's group (1.00 is\nneutral) becomes a bonus <strong>only if it is meaningfully above neutral (at least 1.03)</strong>. Bonus = <code>(IV - 1) x 6</code> (7 for\nharness, 4 for quarter horse), capped at <strong>+5 points</strong>. Post positions that underperform get no penalty, only the\nabsence of a bonus.</p>\n<h3>6.5 Running-style bonus</h3>\n<p>Same idea using the run-style group (E, E/P, P, S): <code>(IV - 1) x 6</code> (7 harness, 5 QH), capped at <strong>+5</strong>, then scaled by\nhow strong the horse's style actually is (its speed points divided by 8, limited to 0 to 1). A horse barely\nfitting \"E\" gets only a fraction of the bonus.</p>\n<h3>6.6 Wet-track bonus</h3>\n<p>Today's track condition is sorted into a wetness tier:</p>\n<div class=\"tw\"><table>\n<thead>\n<tr>\n<th>Tier</th>\n<th>Conditions</th>\n<th>Max bonus</th>\n</tr>\n</thead>\n<tbody>\n<tr>\n<td>0</td>\n<td>Fast, firm, good, etc.</td>\n<td>none</td>\n</tr>\n<tr>\n<td>1</td>\n<td>Yielding, soft</td>\n<td>+0.75</td>\n</tr>\n<tr>\n<td>2</td>\n<td>Sloppy, muddy, heavy, wet</td>\n<td>+1.5</td>\n</tr>\n</tbody>\n</table></div>\n<p>The bonus is scaled by running style (E = 100%, E/P = 65%, P = 25%, S = 0%) and by speed-point strength. Rationale:\non wet tracks, front-runners tend to benefit. This bonus <strong>stacks on top</strong> of the style bonus because it comes from\na different source (today's condition rather than the meet's historical stats).</p>\n<h3>6.7 Final score</h3>\n<pre><code>FINAL SCORE = base + post bonus + style bonus + wet bonus\n</code></pre>\n<p>Scratched horses are forced to a score of 0 at the very end and excluded from everything below.</p>\n<hr />\n<h2>7. Stage 4: From score to probability</h2>\n<p>Scores rank the horses; <strong>probabilities</strong> are needed to put a price on a bet.</p>\n<ol>\n<li><strong>Standardize.</strong> For the live (non-scratched) field, compute each horse's z-score: how many standard deviations\n   above or below the field average it is. Capped at +/-2.5 so one outlier cannot dominate.</li>\n<li><strong>Softmax.</strong> <code>weight = exp(0.70 x z)</code>, then divide each by the field total so they sum to 100%. The <code>0.70</code>\n   (<code>SCORE_PROB_K</code>) sets how sharply a score lead turns into a probability lead.</li>\n<li><strong>Shrink 15% toward an even split.</strong> <code>final = 0.85 x model + 0.15 x (1 / field size)</code>. The model cannot see most of\n   what decides a race (trips, pace scenarios, workouts, equipment, luck), so its raw confidence overstates reality.\n   This one blend makes every downstream number less confident at once.</li>\n</ol>\n<p>Not used here: <strong>live odds are deliberately not blended into win probability.</strong> The score already contains a small\nmarket component (section 5); blending odds in again would count the market twice.</p>\n<h3>Extending to place / show / exacta / trifecta (Harville)</h3>\n<p>The Harville formula assumes that, once a horse is removed, everyone else's chances scale up proportionally:</p>\n<ul>\n<li>Exacta A over B: <code>P(A) x P(B) / (1 - P(A))</code></li>\n<li>Trifecta A-B-C: <code>P(A) x P(B)/(1-P(A)) x P(C)/(1-P(A)-P(B))</code></li>\n<li><strong>Place</strong> (top 2) and <strong>Show</strong> (top 3): the sum of every way that horse can finish in those spots</li>\n</ul>\n<p>These probabilities are the <strong>hit rates</strong> that appear everywhere on the panel.</p>\n<hr />\n<h2>8. Stage 5 and 6: Pricing every bet</h2>\n<p>Every payout comes from <strong>real money in the pools</strong>, not a guessed formula. If the needed pool or probables data has\nnot loaded, that bet is simply not generated.</p>\n<h3>Takeout</h3>\n<p>The track keeps a cut of every pool. The model backs the actual takeout out of the win pool using the displayed\nodds (median across horses, accepted if between 10% and 30%). Otherwise it uses a default 18%. Place and show reuse\nthe same figure.</p>\n<h3>How each bet type is priced</h3>\n<div class=\"tw\"><table>\n<thead>\n<tr>\n<th>Bet</th>\n<th>Payout method</th>\n<th>Probability</th>\n</tr>\n</thead>\n<tbody>\n<tr>\n<td><strong>Win</strong></td>\n<td><code>(win pool x (1 - takeout)) / $ bet on that horse</code></td>\n<td>Win probability</td>\n</tr>\n<tr>\n<td><strong>Place / Show</strong></td>\n<td>Pari-mutuel split: each placing horse's stake is returned, the leftover profit is split <em>equally</em> among the placers, then divided by that horse's own stake. This is why a heavy favorite pays less than a longshot even when both place. Because <em>who else places</em> is unknown, it is priced for every plausible partner set among the model's top 5. The worst case is used for the payout floor, the best case is displayed, and the <strong>average</strong> is used for EV</td>\n<td>Place / Show probability</td>\n</tr>\n<tr>\n<td><strong>Exacta</strong> (straight, wheel, box)</td>\n<td>The <strong>real payouts from the Probables grid</strong>, per combination</td>\n<td>Harville exacta probability per combination, summed across the ticket</td>\n</tr>\n<tr>\n<td><strong>Trifecta box</strong></td>\n<td><strong>A hybrid, the roughest estimate.</strong> Take the real exacta payout for the first two horses and scale it by the model's probability for the third. It is capped by a ceiling derived from the pool size. Low and high estimates are +/-40% of the expected payout</td>\n<td>Harville trifecta probability, summed across all orderings</td>\n</tr>\n</tbody>\n</table></div>\n<h3>Your own bet is counted in the pool</h3>\n<p>A displayed payout reflects the pool as it is <em>without</em> your wager. Placing a bet adds your stake to the pool and to\nyour horse's (or combination's) money, which lowers the real payout. So every bet line is re-priced with <strong>its own\nstake added</strong>: a $6 bet in a $4,000 pool is priced as if the pool were $4,006, and a $250 custom wager is priced as if\nit were $4,250. Each line uses its own stake and bet type.</p>\n<ul>\n<li><strong>Win / Place / Show:</strong> the stake is added to the pool and to that horse's money before the payout is worked out.</li>\n<li><strong>Exacta and trifecta:</strong> the combination's existing money is backed out of the displayed payout, then each combination\n  gets its share of the ticket while the whole ticket total is added to the pool.</li>\n<li>For a small bet in a large pool the change is tiny. It matters when a bet is a real share of the pool, such as a larger\n  custom wager or a small track. Pool movement before post and breakage are not modeled.</li>\n</ul>\n<h3>Expected value</h3>\n<pre><code>EV = (hit probability x payout if it hits) - cost\nEV per $1 = EV / cost\n</code></pre>\n<p>EV per dollar is the main yardstick for comparing bets of different sizes and types.</p>\n<h3>How much to bet (stake sizing)</h3>\n<p>Stakes are limited to <strong>$2 to $6</strong> for single/exacta bets (trifecta boxes have their own $12 ceiling and 50-cent\nbase unit assumption):</p>\n<ul>\n<li><strong>Win / Place / Show:</strong> stake grows with the horse's <em>conviction</em>, which is how far its z-score sits above a\n  \"typical leader\" level of 1.3 (up to a ceiling of 2.5). Reason: the best horse in any field naturally looks about\n  1 to 1.5 standard deviations above average just from ranking statistics, so only a clearly above-typical leader\n  earns more than the $2 minimum.</li>\n<li><strong>Exacta straight:</strong> stake scales with the <em>edge</em> of the live payout versus what the model thinks is fair.</li>\n<li><strong>Wheels, boxes:</strong> fixed at the base unit times the number of combinations.</li>\n</ul>\n<p>Stake size never comes from odds or pool size.</p>\n<p><strong>Custom amounts.</strong> The extension assumes the user bets conservatively with relatively small wagers. While the automatic\nwager options will be within $2 and $6 (up to $12 for trifectas), there is an option to type in a custom amount on the\ncustom wager line. This will calculate your estimated payout using the custom amount and whichever horses you select,\nincluding the effect of your own stake on the pool (see \"Your own bet is counted in the pool\" above).</p>\n<hr />\n<h2>9. Stage 7 and 8: Filtering and the final recommendation</h2>\n<h3>9.1 Candidate list</h3>\n<p>On every race the engine builds <strong>every</strong> candidate it has data for: Win, Place (top 2 horses), Show (top 3 horses),\nExacta Straight, Exacta Wheel (3 and 4 horses), Exacta Box (2 and 3), Trifecta Box (3 and 4).</p>\n<h3>9.2 Eligibility: the near-tie rule</h3>\n<p>A bet is only <em>eligible</em> for recommendation when the model sees a clear dividing line between the horses the ticket\n<strong>includes</strong> and the horses it <strong>leaves out</strong>. Two horses are \"near-tied\" when the weaker one's win probability\nis at least <strong>75%</strong> of the stronger one's (using the pre-shrink probabilities).</p>\n<p>The <strong>boundary rule</strong>: <em>the first horse the bet does not cover must not be near-tied with the weakest horse it does cover.</em></p>\n<div class=\"tw\"><table>\n<thead>\n<tr>\n<th>Bet</th>\n<th>Must NOT be near-tied</th>\n</tr>\n</thead>\n<tbody>\n<tr>\n<td>Win</td>\n<td>#2 vs #1</td>\n</tr>\n<tr>\n<td>Place on a horse</td>\n<td>#3 vs that horse</td>\n</tr>\n<tr>\n<td>Show on a horse</td>\n<td>#4 vs that horse</td>\n</tr>\n<tr>\n<td>Exacta straight</td>\n<td>#3 vs #2 (and #2 vs #1)</td>\n</tr>\n<tr>\n<td>Exacta box (2)</td>\n<td>#3 vs #2</td>\n</tr>\n<tr>\n<td>Exacta box (3)</td>\n<td>#4 vs #3</td>\n</tr>\n<tr>\n<td>Wheel (3 horses)</td>\n<td>#4 vs #3</td>\n</tr>\n<tr>\n<td>Wheel (4 horses)</td>\n<td>#5 vs #4</td>\n</tr>\n<tr>\n<td>Trifecta box (3 / 4)</td>\n<td>#4 vs #3 / #5 vs #4</td>\n</tr>\n</tbody>\n</table></div>\n<p>Straights/wheels need the top horse to stand clear of the rest. Boxes are meant for when the top horses <em>are</em>\nclose, so the two families are mutually exclusive by design.</p>\n<h3>9.3 Value gates</h3>\n<p>From the eligible bets, only those passing <strong>both</strong> survive:</p>\n<ul>\n<li><strong>EV per $1 of at least -15%.</strong> Slightly negative on purpose: racing is high variance, and a high-hit-rate bet with\n  a small modeled shortfall is acceptable. Setting this to 0 would require strictly positive EV.</li>\n<li><strong>Payout floor of 1.4x.</strong> Even the <em>worst-case</em> payout must be at least 1.4x the stake. This screens out bets that\n  are technically fine on EV but only return pennies of profit (heavy favorites to place or show).</li>\n<li><strong>Trifecta only:</strong> also needs a modeled hit rate of at least 8% and a live trifecta pool.</li>\n</ul>\n<h3>9.4 Picking the Value Pick (A) and Safety Pick (B)</h3>\n<ul>\n<li><strong>Value Pick (Option A):</strong> the surviving bet with the highest EV per dollar.</li>\n<li><strong>Safety Pick (Option B):</strong> the surviving bet with the highest hit rate, of a <em>different bet type</em> than the Value Pick.</li>\n</ul>\n<p>These are the first two rows of the Wager Recommendations table, labeled \"Value Pick\" and \"Safety Pick\".</p>\n<h3>9.5 \"No Bet\" outcomes</h3>\n<div class=\"tw\"><table>\n<thead>\n<tr>\n<th>Message</th>\n<th>Meaning</th>\n</tr>\n</thead>\n<tbody>\n<tr>\n<td><strong>No Bet - Too Close</strong></td>\n<td>No bet passed the near-tie rule at all (the field is too evenly matched, or data is missing)</td>\n</tr>\n<tr>\n<td><strong>No Bet - Low Payout</strong></td>\n<td>Eligible bets existed, but none cleared the EV floor or the 1.4x payout floor</td>\n</tr>\n</tbody>\n</table></div>\n<h3>9.6 Everything else on the screen</h3>\n<ul>\n<li><strong>NOT RECOMMENDED rows:</strong> the best candidate of every bet type <em>not</em> used by Option A/B, shown for reference\n  with <strong>all filters ignored</strong>. A way to see the engine's best version of each ticket.</li>\n<li><strong>Custom wager row:</strong> the user types an amount, picks a bet type and horses, and the engine prices it with the\n  same formulas. It is never part of Option A/B.</li>\n</ul>\n<p>All generated bet options are built from the horses the model ranks highest for that bet type, and only the ones that\n  pass a strict value gate are recommended. The gates assume the user doesn't want to risk money for less than <strong>1.4 times\n  their stake</strong> back in the worst case, and doesn't want a bet whose expected value is worse than <strong>-15% per dollar</strong>.\n  Using the custom wager line, you can calculate probability and payouts for any horse that is not scratched, not just\n  those the model designates as good value. The custom line shows the numbers but does not apply the value gate, so it\n  will price a poor-value bet just as readily as a good one.\n- <strong>Low-data warning:</strong> if the highest active score is under 60 (but above 0), a red box warns that whole scoring\n  columns are probably missing (for example, a field of first-time starters). It is a warning only; it does not\n  change any score or block any bet.\n- <strong>Change arrows:</strong> a rank or score change is highlighted for 20 seconds so late shifts (odds moves, scratches)\n  are noticed.\n- <strong>Comments panel:</strong> the TwinSpires positive and negative comments for the top five horses, for human judgment.\n  The model does not read them.</p>\n<hr />\n<h2>10. Glossary</h2>\n<ul>\n<li><strong>EV (expected value):</strong> the average profit per bet if it were placed many times, using the model's hit rate.</li>\n<li><strong>Harville:</strong> formula that turns win probabilities into probabilities for exact finishing orders.</li>\n<li><strong>IV (impact value):</strong> a track-stats figure where 1.00 is neutral and above 1.00 means that group wins more than its share.</li>\n<li><strong>Near-tie:</strong> two horses whose win probabilities are within 25% of each other.</li>\n<li><strong>Pari-mutuel:</strong> betting system where all bets form a pool, the track takes a cut, and winners split the rest.</li>\n<li><strong>Probables:</strong> the live grid of what each exacta combination would pay right now.</li>\n<li><strong>Takeout:</strong> the percentage of each pool the track and state keep before paying winners.</li>\n<li><strong>z-score:</strong> how many standard deviations a value is above or below the group average.</li>\n</ul></main></body></html>";
